虚拟滚动

动态尺寸虚拟列表 AsamDynamicSizeList

在有限视窗内渲染可见项目，减少长列表的挂载量。

使用场景 ​

在有限视窗内渲染可见项目，减少长列表的挂载量。

契约与组合 ​

这是虚拟列表基础入口，只渲染可见范围与 cache 缓冲。提供固定 height / width 和 total / data，默认插槽的 style 必须放在真实项目根节点上，否则绝对位置与尺寸计算无法生效。插槽提供 index / style / data / isScrolling。

itemSize 是尺寸函数；estimatedItemSize 用于尚未测算范围。列表未提供 resetAfterIndex 公开方法；改变尺寸函数结果时，以 key 重建尺寸缓存并用 scrollTo 恢复原滚动位置，边界示例完整演示这一流程。

通过公开方法定位列表，支持水平或垂直布局。 完整说明。

这些低层入口没有表格列注册或独立选择模型，也没有额外四档 size；行列尺寸由宿主决定，颜色、间距和字体来自 ASAM tokens。

基础用法与状态

vue
<template>
	<AsamDynamicSizeList
		v-bind="{
			height: 200,
			width: 280,
			total: items.length,
			data: items,
			itemSize,
			estimatedItemSize: 50
		}"
	>
		<template #default="{ index, data, style }">
			<div :style="style" class="virtual-cell">{{ data[index] }}</div>
		</template>
	</AsamDynamicSizeList>
</template>

defineOptions({ name: 'AsamDynamicSizeListBasicDemo' });
import { AsamDynamicSizeList } from '@asam-ui/ui';

const itemSize = (index: number): number => (index % 2 === 0 ? 40 : 60);
</script>

<template>
	<div class="demo-root-layout">
		<AsamButton native-type="submit" type="default" @click="list?.scrollToItem(99, 'end')"
			>到最后一项</AsamButton
		>
		<AsamButton native-type="submit" type="default" @click="list?.resetScrollTop()"
			>返回起点</AsamButton
		>
		<AsamButton native-type="submit" type="default" @click="resize">改变项目尺寸</AsamButton>
		<AsamCheckboxItem v-model="horizontal">水平布局</AsamCheckboxItem>
		<AsamCheckboxItem v-model="rtl">RTL</AsamCheckboxItem>
		<AsamCheckboxItem v-model="always">滚动条持续显示</AsamCheckboxItem>
		<p data-testid="virtual-events">范围 {{ range }}；滚动 {{ eventState }}；边界 {{ edge }}</p>
		<AsamDynamicSizeList
			:key="[revision, horizontal, rtl].join(':')"
			ref="list"
			:data="items"
			:height="200"
			:width="320"
			:total="items.length"
			:item-size="itemSize"
			:estimated-item-size="50"
			:layout="horizontal ? 'horizontal' : 'vertical'"
			:direction="rtl ? 'rtl' : 'ltr'"
			:scrollbar-always-on="always"
			:use-is-scrolling="true"
			@item-rendered="onRendered"
			@scroll="onScroll"
			@end-reached="edge = $event"
		>
			<template #default="{ index, style, isScrolling }">
				<div :style="style" class="virtual-cell" :data-index="index">
					{{ items[index] }}{{ isScrolling ? '（滚动中）' : '' }}
				</div>
			</template>
		</AsamDynamicSizeList>
	</div>
</template>
<script lang="ts" setup>
defineOptions({ name: 'AsamDynamicSizeListBoundaryDemo' });
import { nextTick, ref } from 'vue';
import { AsamCheckboxItem, AsamButton, AsamDynamicSizeList } from '@asam-ui/ui';
import type { AsamDynamicSizeListExpose, AsamDynamicSizeListEmits } from '@asam-ui/ui';
const list = ref<AsamDynamicSizeListExpose>();
const items = Array.from({ length: 100 }, (_, index) => '记录 ' + (index + 1));
const horizontal = ref(false);
const rtl = ref(false);
const always = ref(true);
const revision = ref(0);
const eventState = ref('');
const edge = ref('');
const onScroll = (...[direction, offset, requested]: AsamDynamicSizeListEmits['scroll']): void => {
	eventState.value = direction + ':' + offset + ':' + requested;
};
const onRendered = (
	...[start, end, first, last]: AsamDynamicSizeListEmits['itemRendered']
): void => {
	range.value = start + ':' + end + ':' + first + ':' + last;
};
const range = ref('');
const large = ref(false);
const itemSize = (index: number): number => (index % 2 ? 60 : 40) + (large.value ? 30 : 0);
async function resize(): Promise<void> {
	const offset = list.value?.states.scrollOffset ?? 0;
	large.value = !large.value;
	revision.value++;
	await nextTick();
	list.value?.scrollTo(offset);
}
</script>
