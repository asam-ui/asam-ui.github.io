字典虚拟选择器 ​

在虚拟选择器上管理字典选项、重载和请求取消。

基础用法 ​

示例通过本地 provider 获取字典，支持模型更新、暂停与重载。所有请求都留在演示内部。

点击“读取当前标签”读取实例的只读 selectedLabel：空模型返回空字符串，选择后返回当前标签。focus() 与 blur() 为无参数方法，仅控制输入焦点，不改写模型；失焦按钮阻止鼠标按下时抢走焦点，以便观察 blur() 的实际作用。勾选“启用多选”切换到独立数组模型；空选择读取空数组，选择多个选项后返回标签数组，输入焦点方法仍不改写模型。

本地创建与字典规范项 ​

allow-create 与 filterable 开启后，createOption(text) 同步返回完整 AsamOption<V, I>， 包含 label、正确类型的 value 和真实 raw。默认记录使用字符串 value；数字、布尔值 或自定义 raw 必须提供类型一致的同步工厂。

<template>
	<section class="data-demo">
		<div class="actions">
			<AsamButton type="primary" native-type="button" @click="reload">强制重载</AsamButton>
			<AsamButton type="primary" native-type="button" @click="control?.clearCreatedOptions()"
				>清空本地创建集合</AsamButton

				>清空模型</AsamButton
			>
			<AsamButton
				type="primary"
				native-type="button"
				@click="tenant = tenant === 'a' ? 'b' : 'a'"
				>切换逻辑来源</AsamButton
			>
			<AsamCheckboxItem v-model="failure">模拟失败</AsamCheckboxItem>
			<AsamCheckboxItem v-model="canonical">来源接受 Omega 同值项</AsamCheckboxItem>
		</div>
		<AsamDictSelectV2
			ref="control"
			v-model="value"
			:dict-type="tenant"
			:create-option="createOption"
			value-mode="object"
			allow-create
			filterable
			clearable
			size="default"
			placeholder="输入 Omega 并选择创建项"
			@success="providerValues = $event.map((item) => item.value)"
			@error="message = $event.message"
		>
			<template #header><p>本地创建和来源选项分别保存</p></template>
			<template #footer><p>刷新保留本地项，逻辑来源改变时清空本地集合</p></template>
		</AsamDictSelectV2>
		<p role="status">{{ result }} {{ message }}</p>
		<pre>模型：{{ JSON.stringify(value) }}</pre>
		<pre>工厂调用：{{ JSON.stringify(createdTexts) }}</pre>
		<pre>provider success 值：{{ JSON.stringify(providerValues) }}</pre>
	</section>
</template>

<script lang="ts" setup>
import { inject, nextTick, ref } from 'vue';
import { AsamCheckboxItem, AsamButton, AsamDictSelectV2 } from '@asam-ui/ui';
import type {
	AsamOption,
	AsamCreateOption,
	AsamValue,
	AsamRemoteSelectV2Expose
} from '@asam-ui/ui';
import type { DemoState, Member } from './provider.js';
const state = inject<DemoState>('selection-local-demo');
if (!state) throw new Error('Demo.vue 必须先配置演示 provider。');
const { tenant, failure, canonical } = state;
defineOptions({ name: 'AsamDictSelectV2LocalCreationChild' });
const value = ref<AsamOption<string, Member> | null>(null);
const control = ref<AsamRemoteSelectV2Expose<string, Member>>();
const createdTexts = ref<string[]>([]);
const providerValues = ref<AsamValue[]>([]);
const message = ref('');
const result = ref('');
const createOption: AsamCreateOption<string, Member> = (text) => {
	createdTexts.value.push(text);
	return { label: text, value: text, raw: { id: 'local-' + text, label: text, value: text } };
};

async function reload() {
	await nextTick();
	result.value = (await control.value?.reload({ force: true }))?.status ?? 'cancelled';
	if (result.value === 'success') message.value = '';
}
</script>

import { ref } from 'vue';
import type { AsamDataProvider, AsamDictEntry } from '@asam-ui/ui';
export interface Member extends AsamDictEntry {
	readonly id: string;
	readonly value: string;
}
export function createDemoState() {
	return { tenant: ref('a'), failure: ref(false), canonical: ref(false) };
}
export type DemoState = ReturnType<typeof createDemoState>;
export function createDemoProvider(state: DemoState): AsamDataProvider {
	return {
		dict: {
			async refresh(context) {
				context.signal.throwIfAborted();

数据来源与覆盖 ​

dictType 与 dataSource 只选一种；dataSource 使用 defineAsamDictSource 创建，不能再传 dictValueType。短来源的 dictValueType 可将字典值转换为 string、number 或 boolean。应用通过 provideAsamData 注入 dict.get，演示在独立子应用中使用本地字典，切换语言或离开时卸载。

生命周期、事件与实例 ​

active=false 暂停请求并保留上次成功数据；reload({ force: true }) 请求新数据，cancel() 取消当前请求。success 返回规范化选项数组，error 返回 AsamDataError；旧成功数据更新失败时组件显示过期状态。reload 的 status 为 success、error 或 cancelled；不能把取消当作成功。

immediate 控制初始化是否请求，active 控制请求是否启用。初始化和默认值以本页自动 API 的原始来源为准。

import { ref } from 'vue';
import type { AsamDataProvider } from '@asam-ui/ui';
export interface Member {
	readonly label: string;
	readonly value: number;
	readonly id: number;
	readonly team: string;
	readonly disabled?: boolean;
}
const rows: readonly Member[] = [
	{ label: 'Alpha', value: 1, id: 1, team: '一组' },
	{ label: 'Beta', value: 2, id: 2, team: '一组', disabled: true },
	{ label: 'Gamma', value: 3, id: 3, team: '二组' },
	{ label: 'Delta', value: 4, id: 4, team: '二组' },
	{ label: 'Epsilon', value: 5, id: 5, team: '二组' }
];
export function createDemoState() {
	const held = ref(false),
		empty = ref(false),
		failure = ref(false),
		calls = ref(0);
	const pending = new Set<() => void>();
	async function read(signal: AbortSignal, text: string) {
		signal.throwIfAborted();
		calls.value++;
		if (held.value)
			await new Promise<void>((done, reject) => {
				const finish = () => {
					pending.delete(finish);
					signal.removeEventListener('abort', abort);
					done();
				};
				const abort = () => {
					pending.delete(finish);
					signal.removeEventListener('abort', abort);
					reject(signal.reason);
				};
				pending.add(finish);
				signal.addEventListener('abort', abort, { once: true });
				if (signal.aborted) abort();
			});
		signal.throwIfAborted();
		if (failure.value) throw new Error('演示选择器请求失败');
		return empty.value
			? []
			: rows.filter((item) => item.label.toLowerCase().includes(text.toLowerCase()));
	}
	function release() {
		held.value = false;
		for (const finish of [...pending]) finish();
	}
	return { held, empty, failure, calls, read, release };
}
export type DemoState = ReturnType<typeof createDemoState>;
export function createDemoProvider(state: DemoState): AsamDataProvider {
	return {
		dict: {
			async refresh(context) {
				context.signal.throwIfAborted();
			},
			async get(type, context) {
				if (type !== 'presentation') throw new Error('Unknown local dictionary');
				return (await state.read(context.signal, '')).map((item) => ({ ...item }));
			}
		}
	};
}

	const held = ref(false),
		empty = ref(false),
		failure = ref(false),
		calls = ref(0);
	const pending = new Set();
	async function read(signal, text) {
		signal.throwIfAborted();
		calls.value++;
		if (held.value)
			await new Promise((done, reject) => {
				const finish = () => {
					pending.delete(finish);
					signal.removeEventListener('abort', abort);
					done();
				};
				const abort = () => {
					pending.delete(finish);
					signal.removeEventListener('abort', abort);
					reject(signal.reason);
				};
				pending.add(finish);
				signal.addEventListener('abort', abort, { once: true });
				if (signal.aborted) abort();
			});
		signal.throwIfAborted();
		if (failure.value) throw new Error('演示选择器请求失败');
		return empty.value
			? []
			: rows.filter((item) => item.label.toLowerCase().includes(text.toLowerCase()));
	}
	function release() {
		held.value = false;
		for (const finish of [...pending]) finish();
	}
	return { held, empty, failure, calls, read, release };
}

局部请求错误插槽 ​

可选 error 插槽仅在当前请求错误存在时调用，接收只读 { error, stale, retry }。 error 与本实例 error 事件中的 AsamDataError 为同一对象，保留 cause、kind、 message 与 notified；渲染插槽不会再次通知 provider。stale 表示保留的数据已过期。

retry() 不接收参数，直接返回现有 reload({ force: true }) 的 Promise，结果为 AsamReloadResult<readonly AsamOption<V, I>[]> 的 success、error 或 cancelled 分支。 重试遵守 active、当前来源、取消和卸载规则，不改写模型。字典强刷沿用 provider 的 refresh 后 get；缺少 refresh 仍是原配置错误。

提供该插槽时，自有内容替换整个默认局部错误区；返回空内容也表示主动替换。 未提供时继续显示原错误文字、过期提示和默认重试按钮。无 error 的单独 stale 状态 不调用此插槽；empty、loading 和其他基础插槽继续使用原作用域。

以下完整 TS/JS 示例先制造失败，再等待插槽 retry 的返回结果；成功后可再次制造 更新失败，观察旧结果与模型保留。每次挂载使用独立子 app 和本地 provider。
