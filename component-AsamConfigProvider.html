全局配置 AsamConfigProvider

向子树提供组件配置和 ASAM 控件密度。

使用场景 ​

向子树提供组件配置和 ASAM 控件密度。

基础用法与状态 ​

下列示例只使用本入口的实际公开能力。

基础用法与状态

展开源码
vue
<template>
	<label>
		<span>继承密度</span>
		<AsamSelect v-model="size">
			<AsamOption v-for="value in sizes" :key="value" :label="value" :value="value">{{
				value
			}}</AsamOption>
		</AsamSelect>
	</label>
	<AsamConfigProvider :size="size">
		<div class="actions">
			<AsamTag>继承 {{ size }}</AsamTag>
			<AsamTag size="small">显式 small 优先</AsamTag>
		</div>
	</AsamConfigProvider>
</template>

<script lang="ts" setup>
defineOptions({ name: 'AsamConfigProviderBasicDemo' });
import { ref } from 'vue';
import { AsamSelect, AsamOption, AsamConfigProvider, AsamTag } from '@asam-ui/ui';
import type { AsamControlSize } from '@asam-ui/ui';
const sizes: AsamControlSize[] = ['large', 'big', 'default', 'small'];
const size = ref<AsamControlSize>('default');
</script>

<style scoped>
.actions {
	display: flex;
	flex-wrap: wrap;
	gap: var(--asam-space-md);
	align-items: center;
}
</style>

使用约定 ​

size 接受 large/big/default/small，省略时保留默认密度；配置变化保持响应式。子控件显式 size 优先于最近 group、Form、ConfigProvider。locale、namespace、zIndex、button、message 等配置以公开 Props 为准；它提供上下文，不直接请求业务数据。default 插槽提供真实配置上下文，无业务事件。

	alignCenter: boolean;
	draggable: boolean;
	overflow: boolean;
	transition?: NonNullable<AsamConfigProviderProps['dialog']>['transition'];
};

	draggable: false,
	overflow: false

const enableTransition = ref(false);
const isObjectTransition = ref(false);

const ANIMATION_DURATION = 300;

const globalConfig = computed<GlobalConfig>(() => {
	let transition: NonNullable<AsamConfigProviderProps['dialog']>['transition'] | undefined;
	if (enableTransition.value) {
		if (isObjectTransition.value) {
			transition = {
				css: false,
				onBeforeEnter(el) {
					nextTick(() => {
						if (buttonRef.value) {
							const buttonRect = buttonRef.value.ref!.getBoundingClientRect();
							const dialogEl = el.querySelector('.el-dialog') as HTMLElement;

							if (dialogEl) {
								const dialogRect = dialogEl.getBoundingClientRect();

								const offsetX =
									buttonRect.left +
									buttonRect.width / 2 -
									(dialogRect.left + dialogRect.width / 2);
								const offsetY =
									buttonRect.top +
									buttonRect.height / 2 -
									(dialogRect.top + dialogRect.height / 2);

								dialogEl.style.transform = `translate(${offsetX}px, ${offsetY}px) scale(0.3)`;
								dialogEl.style.opacity = '0';
							}
						}
					});
				},
				onEnter(el, done) {
					nextTick(() => {
						const dialogEl = el.querySelector('.el-dialog') as HTMLElement;
						if (dialogEl) {
							// force reflow
							void dialogEl.offsetHeight;

							dialogEl.style.transition = `all ${ANIMATION_DURATION}ms cubic-bezier(0.4, 0, 1, 1)`;
							dialogEl.style.transform = 'translate(0, 0) scale(1)';
							dialogEl.style.opacity = '1';

							// wait for animation to complete, then cleanup inline styles to avoid affecting drag
							setTimeout(() => {
								dialogEl.style.transition = '';
								dialogEl.style.transform = '';
								dialogEl.style.opacity = '';
								done();
							}, ANIMATION_DURATION);
						} else {
							done();
						}
					});
				},
				onLeave(el, done) {
					const dialogEl = el.querySelector('.el-dialog') as HTMLElement;
					if (dialogEl) {
						if (buttonRef.value) {
							const buttonRect = buttonRef.value.ref!.getBoundingClientRect();
							const dialogRect = dialogEl.getBoundingClientRect();

							const currentTransform = dialogEl.style.transform;
							let dragOffsetX = 0;
							let dragOffsetY = 0;

							// avoid draggable effect
							if (currentTransform) {
								const translateMatch = currentTransform.match(
									/translate\(([^,]+),\s*([^)]+)\)/

								if (translateMatch) {
									dragOffsetX = Number.parseFloat(translateMatch[1]);
									dragOffsetY = Number.parseFloat(translateMatch[2]);
								}
							}

							const offsetX =
								buttonRect.left +
								buttonRect.width / 2 -
								(dialogRect.left + dialogRect.width / 2) +
								dragOffsetX;
							const offsetY =
								buttonRect.top +
								buttonRect.height / 2 -
								(dialogRect.top + dialogRect.height / 2) +
								dragOffsetY;

							dialogEl.style.transition = `all ${ANIMATION_DURATION}ms cubic-bezier(0.4, 0, 1, 1)`;
							dialogEl.style.transform = `translate(${offsetX}px, ${offsetY}px) scale(0.3)`;

							// wait for animation to complete, then cleanup inline styles
							setTimeout(() => {
								dialogEl.style.transition = '';
								dialogEl.style.transform = '';
								dialogEl.style.opacity = '';
								done();
							}, ANIMATION_DURATION);
						} else {
							done();
						}
					} else {
						done();
					}
				}
			};
		} else {
			transition = 'dialog-bounce';
		}
	}
	return {

	overflow: false

const globalConfig = computed(() => {
	let transition;
	if (enableTransition.value) {
		if (isObjectTransition.value) {
			transition = {
				css: false,
				onBeforeEnter(el) {
					nextTick(() => {
						if (buttonRef.value) {
							const buttonRect = buttonRef.value.ref.getBoundingClientRect();
							const dialogEl = el.querySelector('.el-dialog');
							if (dialogEl) {
								const dialogRect = dialogEl.getBoundingClientRect();
								const offsetX =
									buttonRect.left +
									buttonRect.width / 2 -
									(dialogRect.left + dialogRect.width / 2);
								const offsetY =
									buttonRect.top +
									buttonRect.height / 2 -
									(dialogRect.top + dialogRect.height / 2);
								dialogEl.style.transform = `translate(${offsetX}px, ${offsetY}px) scale(0.3)`;
								dialogEl.style.opacity = '0';
							}
						}
					});
				},
				onEnter(el, done) {
					nextTick(() => {
						const dialogEl = el.querySelector('.el-dialog');
						if (dialogEl) {
							// force reflow
							void dialogEl.offsetHeight;
							dialogEl.style.transition = `all ${ANIMATION_DURATION}ms cubic-bezier(0.4, 0, 1, 1)`;
							dialogEl.style.transform = 'translate(0, 0) scale(1)';
							dialogEl.style.opacity = '1';
							// wait for animation to complete, then cleanup inline styles to avoid affecting drag
							setTimeout(() => {
								dialogEl.style.transition = '';
								dialogEl.style.transform = '';
								dialogEl.style.opacity = '';
								done();
							}, ANIMATION_DURATION);
						} else {
							done();
						}
					});
				},
				onLeave(el, done) {
					const dialogEl = el.querySelector('.el-dialog');
					if (dialogEl) {
						if (buttonRef.value) {
							const buttonRect = buttonRef.value.ref.getBoundingClientRect();
							const dialogRect = dialogEl.getBoundingClientRect();
							const currentTransform = dialogEl.style.transform;
							let dragOffsetX = 0;
							let dragOffsetY = 0;
							// avoid draggable effect
							if (currentTransform) {
								const translateMatch = currentTransform.match(
									/translate\(([^,]+),\s*([^)]+)\)/
								);
								if (translateMatch) {
									dragOffsetX = Number.parseFloat(translateMatch[1]);
									dragOffsetY = Number.parseFloat(translateMatch[2]);
								}
							}
							const offsetX =
								buttonRect.left +
								buttonRect.width / 2 -
								(dialogRect.left + dialogRect.width / 2) +
								dragOffsetX;
							const offsetY =
								buttonRect.top +
								buttonRect.height / 2 -
								(dialogRect.top + dialogRect.height / 2) +
								dragOffsetY;

							setTimeout(() => {
								dialogEl.style.transition = '';
								dialogEl.style.transform = '';
								dialogEl.style.opacity = '';
								done();
							}, ANIMATION_DURATION);
						} else {
							done();
						}
					} else {
						done();
					}
				}
			};
		} else {
			transition = 'dialog-bounce';
		}
	}
	return {
