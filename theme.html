亮暗主题 ​

TS ​

原生 error 状态跟随当前 danger 语义色。

primary 原生 RGB success 原生 RGB warning 原生 RGB danger 原生 RGB error 原生 RGB info 原生 RGB

RGB tuple 保留 Element Plus 编译值，亮暗切换不会将它们改成当前语义色的 RGB。

按 Tab 检查焦点；主题作用于此容器。body 中的弹层使用全局主题。

vue
<template>
	<section ref="scope" class="appearance" :style="{ colorScheme: dark ? 'dark' : 'light' }">

		}}</AsamButton>

		<AsamButton type="default" disabled>禁用操作</AsamButton>

		<p class="theme-error">原生 error 状态跟随当前 danger 语义色。</p>
		<div class="rgb-probes">
			<span
				v-for="type in nativeRgbTypes"
				:key="type"
				:style="{ backgroundColor: `rgba(var(--el-color-${type}-rgb), 0.16)` }"
			>
				{{ type }} 原生 RGB
			</span>
		</div>
		<p>RGB tuple 保留 Element Plus 编译值，亮暗切换不会将它们改成当前语义色的 RGB。</p>

<template>
	<section ref="scope">
		<label>
			<span>big 的 FormItem 间距</span>
			<AsamInputNumber v-model.number="gap" :min="0" />
		</label>
		<label>
			<AsamCheckboxItem v-model="legacy" />
			<span>旧完整主题：省略新增间距字段，使用 fieldGap</span>
		</label>
		<div v-for="size in sizes" :key="size" class="density-row">
			<AsamConfigProvider :size="size">
				<AsamButton type="default">{{ size }}</AsamButton>
				<AsamInput v-model="value" :aria-label="`${size} 输入`" />
				<AsamBaseForm>
					<AsamFormItem :label="size"><AsamInput v-model="value" /></AsamFormItem>
					<AsamFormItem label="下一字段"><AsamInput v-model="value" /></AsamFormItem>
				</AsamBaseForm>
			</AsamConfigProvider>
		</div>
	</section>
</template>

<script setup lang="ts">
import { ref, watchEffect } from 'vue';
import {
	AsamInputNumber,
	AsamCheckboxItem,
	AsamBaseForm,
	AsamButton,
	AsamConfigProvider,
	AsamFormItem,
	AsamInput,
	asamControlSizes
} from '@asam-ui/ui';
import { applyAsamTheme, createAsamTheme } from '@asam-ui/ui/tokens';

const sizes = asamControlSizes;
const scope = ref<HTMLElement>();
const gap = ref(20);
const legacy = ref(false);
const value = ref('四档控件');
watchEffect(() => {
	const theme = createAsamTheme({ density: { big: { formItemMarginBottom: `${gap.value}px` } } });
	if (legacy.value) {
		delete theme.border;
		delete theme.typography.lineHeight;
		for (const density of Object.values(theme.density ?? {}))
			delete density.formItemMarginBottom;
		theme.layout.fieldGap = `${gap.value}px`;
	}
	if (scope.value) applyAsamTheme(theme, scope.value);
});
defineOptions({ name: 'ThemeDensityDemo' });
</script>

<style scoped>
.density-row {
	margin-top: var(--asam-space-md);
}
</style>
