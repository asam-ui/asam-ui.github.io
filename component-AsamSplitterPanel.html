使用约定 ​

size、min、max 使用实际像素值或百分比，v-model:size 接收更新；resizable 和 collapsible 控制调整与折叠。default 插槽放面板内容，必须在 Splitter 内使用。size 是布局尺寸，不能使用控件密度 big。

组合用法 ​

在 AsamSplitter 的完整场景中使用本入口；包含 basic、vertical、collapsible、disable-drag、size、lazy。

<template>
	<section class="component-project" data-demo-host>
		<label
			>演示场景<AsamSelect
				:model-value="selected"
				aria-label="演示场景"
				@update:model-value="selectScenario"
				><AsamOption
					v-for="option in options"
					:key="option.value"
					:value="option.value"
					:label="option.label" /></AsamSelect></label
		><component :is="current" :key="selected" />
	</section>

<script lang="ts" setup>
import { computed, defineAsyncComponent, shallowRef } from 'vue';
import { AsamSelect, AsamOption } from '@asam-ui/ui';
const options = [
	{ value: 'basic/ts', label: '基础用法与状态 (TS)' },
	{ value: 'basic/js', label: '基础用法与状态 (JS)' }
] as const;
const scenarios = {
	'basic/ts': defineAsyncComponent(() => import('./examples/basic/ts/Demo.vue')),
	'basic/js': defineAsyncComponent(() => import('./examples/basic/js/Demo.vue'))
};
const selected = shallowRef<keyof typeof scenarios>('basic/ts');
function selectScenario(value: unknown) {
	const option = options.find((option) => option.value === value);
	if (option) selected.value = option.value;
}
const current = computed(() => scenarios[selected.value]);
</script>
