import { AsamConfigProvider as WebsiteAsamConfigProvider } from '@asam-ui/ui';
import { computed as websiteComputed, inject as websiteInject } from 'vue';
import type { Ref as WebsiteRef } from 'vue';
const websiteFullWidth = websiteInject<WebsiteRef<boolean> | null>(
	Symbol.for('asam.website.preview.fullWidth'),
	null

	websiteFullWidth ? (websiteFullWidth.value ? 'full' : 'contained') : undefined

function websiteSetCrudWidth(width: 'full' | 'contained') {
	if (websiteFullWidth) websiteFullWidth.value = width === 'full';

	typeof document !== 'undefined' && !!document.getElementById('preview-settings-extra');

import { AsamSelect, AsamOption, AsamCheckboxItem, AsamInputNumber } from '@asam-ui/ui';

import { AsamButton } from '@asam-ui/ui';

import { computed, onBeforeUnmount, ref } from 'vue';

import type { AsamControlSize } from '@asam-ui/ui';
import { AsamCrudPage, useAsamCrudController } from '@asam-ui/pro/templates/crud';

	createAsamWriteCoordinator,
	createAsamTableOperationCoordinator

import { crudConfirmation, crudSchema, useCrudLocale } from './I18n.js';
import { createDemoService, queryCodec, editCodec, initialQuery, isKey } from './Service.js';

const { i18n, text } = useCrudLocale();
const size = ref<AsamControlSize>('default');
const permission = ref(true),
	readPermission = ref(true),
	readOnly = ref(false),
	disabled = ref(new URLSearchParams(location.search).has('disabled')),
	scope = ref<'边界A' | '边界B'>('边界A');
const demo = createDemoService(text);
const delayDraft = ref<number | null | undefined>(demo.controls.delay);
const delayControl = computed({
	get: () => delayDraft.value,
	set: (value: number | null | undefined) => {
		delayDraft.value = value;
		demo.controls.delay = value ?? 0;
	}
});
const runtime = createAsamDataRuntime();
runtime.setScope(scope.value);
let sequence = 0;
const crud = useAsamCrudController<Row, Query, Edit, string | number>({
	runtime,
	writes: createAsamWriteCoordinator(),
	operations: createAsamTableOperationCoordinator(),
	rowKey: (row) => row.id,
	isKey,
	dataset: () => scope.value,
	initial: { ...initialQuery },
	...crudSchema(text),
	queryCodec,
	editCodec,
	services: demo.services,
	detail: { mode: 'load', load: demo.detail },
	preferences: {
		scope: () => ({
			application: 'crud-boundary',
			principal: scope.value,
			table: 'assets',
			version: '1'
		}),
		adapter: { load: async () => null, save: async () => {}, reset: async () => {} }
	},
	pageSize: 20,
	disabled,
	readonly: readOnly,
	allowed: (action) => (action === 'detail' ? readPermission.value : permission.value),
	allowedRemove: () => permission.value,
	...crudConfirmation(text, () => i18n.locale.value),
	operationId: (action) => `boundary-${action}-${++sequence}`
});
function switchScope() {
	scope.value = scope.value === '边界A' ? '边界B' : '边界A';
	runtime.setScope(scope.value);
	crud.list.resumeScope({ ...initialQuery });
	if (!disabled.value) void crud.list.submit();
}
let scopeTimer: ReturnType<typeof setTimeout> | undefined;
function delayedScope() {
	clearTimeout(scopeTimer);
	scopeTimer = setTimeout(switchScope, 1000);
}
onBeforeUnmount(() => clearTimeout(scopeTimer));

function updateSize(value: unknown) {
	if (value === 'small' || value === 'default' || value === 'big' || value === 'large')
		size.value = value;
}

function updateDemoControlsSave(value: unknown) {
	if (
		value === 'success' ||
		value === 'failure' ||
		value === 'field-error' ||
		value === 'unknown-applied' ||
		value === 'unknown-not-applied'
	)
		demo.controls.save = value;
}

function updateDemoControlsRemove(value: unknown) {
	if (
		value === 'success' ||
		value === 'failure' ||
		value === 'unknown-applied' ||
		value === 'unknown-not-applied'
	)
		demo.controls.remove = value;
}
</script>

export function createDemoService(text: CrudText = (value) => value) {
	const controls = reactive({
		delay: 100,
		listError: false,
		loadError: false,
		missing: false,
		forbidden: false,
		save: 'success' as Failure,
		remove: 'success' as Failure,
		partial: false,
		verifyUnknown: false,
		refreshError: false,
		prefix: '资产'
	});
	const metrics = reactive({
		list: 0,
		load: 0,
		defaults: 0,
		create: 0,
		update: 0,
		remove: 0,
		verify: 0,
		verifyRemove: 0,
		events: [] as {
			name: string;
			scope: string;
			operationId?: string;
			key?: AsamCrudKey;
			start: number;
			end: number;
		}[]
	});
	const stores = new Map<string, Map<AsamCrudKey, Row>>();
	const writes = new Map<string, boolean>();
	const deletions = new Map<string, boolean>();
	let sequence = 100;
	function store(scope: string) {
		if (!stores.has(scope))
			stores.set(
				scope,
				new Map(
					Array.from({ length: 100 }, (_, i) => {
						const id = i === 1 ? '0' : i;
						return [
							id,
							{
								id,
								version: 1,
								name: `资产${i}`,
								team: i % 2 ? '研发' : '运营',
								status: '启用',
								owner: `成员${i % 5}`,
								email: `member${i}@example.test`,
								note: `合成记录${i} / ${scope}`
							}
						];
					})
				)
			);
		return stores.get(scope)!;
	}
	async function wait(name: string, scope: string, operationId?: string, key?: AsamCrudKey) {
		const event = { name, scope, operationId, key, start: performance.now(), end: 0 };
		await new Promise((resolve) => setTimeout(resolve, controls.delay));
		event.end = performance.now();
		metrics.events.push(event);
	}
	const identity = (scope: string, id: string, key?: AsamCrudKey) =>
		JSON.stringify([scope, id, typeof key, key]);
	async function save(
		model: Readonly<Edit>,
		scope: string,
		operationId: string,
		id?: AsamCrudKey
	) {
		const failure = controls.save;
		await wait(id === undefined ? 'create' : 'update', scope, operationId, id);
		if (failure === 'field-error')
			return {
				status: 'failure' as const,
				message: text('服务字段校验失败'),
				issues: [{ key: 'name', message: text('名称已被占用，请修改名称') }]
			};
		if (failure === 'failure')
			return { status: 'failure' as const, message: text('服务明确拒绝，未写入') };
		const applied = failure !== 'unknown-not-applied';
		if (applied) {
			const key = id ?? ++sequence;
			store(scope).set(key, {
				...model,
				id: key,
				version: (store(scope).get(key)?.version ?? 0) + 1
			});
		}
		writes.set(identity(scope, operationId), applied);
		if (controls.refreshError) controls.listError = true;
		return failure.startsWith('unknown')
			? { status: 'unknown' as const, message: text('响应丢失，请核实服务器结果') }
			: { status: 'success' as const, data: { id } };
	}
	const services: AsamCrudServices<Row, Query, Edit, AsamCrudKey> = {
		async list(request, context) {
			metrics.list++;
			const snapshot = [...store(context.scope).values()].map((row) => ({ ...row }));
			await wait('list', context.scope);
			if (controls.listError) throw new Error(text('受控列表失败'));
			const rows = snapshot.filter((row) =>
				Object.entries(request.query).every(
					([key, value]) => !value || String(Reflect.get(row, key)).includes(value)
				)
			);
			if (request.sort) {
				const { field, order } = request.sort;
				rows.sort(
					(a, b) =>
						String(Reflect.get(a, field)).localeCompare(
							String(Reflect.get(b, field)),
							'zh-CN',
							{ numeric: true }
						) * (order === 'ascending' ? 1 : -1)
				);
			}
			return {
				list: rows
					.slice(
						(request.currentPage - 1) * request.pageSize,
						request.currentPage * request.pageSize
					)
					.map((row) => ({ ...row, name: row.name.replace('资产', controls.prefix) })),
				total: rows.length
			};
		},
		async defaults(context) {
			metrics.defaults++;
			await wait('defaults', context.scope);
			return { name: '', team: '研发', status: '启用', owner: '成员0', email: '', note: '' };
		},
		async load(key, context) {
			metrics.load++;
			await wait('load', context.scope, undefined, key);
			if (controls.loadError) throw new Error(text('受控编辑加载失败'));
			const row = store(context.scope).get(key);
			if (!row) throw new Error(text('记录已不存在'));
			return { ...row };
		},
		create(model, operation, context) {
			metrics.create++;
			return save(model, context.scope, operation.target.operationId);
		},
		update(key, model, operation, context) {
			metrics.update++;
			return save(model, context.scope, operation.target.operationId, key);
		},
		async verify(operation, context) {
			metrics.verify++;
			await wait('verify', operation.scope, operation.target.operationId);
			if (controls.verifyUnknown) return { status: 'unknown' };
			return writes.get(identity(operation.scope, operation.target.operationId))
				? { status: 'success', data: { verified: true } }
				: { status: 'not-applied' };
		},
		async resolve(keys, context) {
			return keys.map((key) => {
				const row = store(context.scope).get(key);
				return row
					? {
							key,
							target: {
								key,
								version: String(row.version),
								payload: { name: row.name }
							}
						}
					: { key, reason: text('记录不存在') };
			});
		},
		async remove(target, snapshot) {
			metrics.remove++;
			const failure = controls.partial && target.key === '0' ? 'failure' : controls.remove;
			await wait('remove', snapshot.context.scope, snapshot.operationId, target.key);
			if (failure === 'failure' || failure === 'field-error')
				return { key: target.key, status: 'failure', message: text('明确未应用，可重试') };
			const applied = failure !== 'unknown-not-applied';
			if (applied) store(snapshot.context.scope).delete(target.key);
			deletions.set(
				identity(snapshot.context.scope, snapshot.operationId, target.key),
				applied
			);
			if (controls.refreshError) controls.listError = true;
			return {
				key: target.key,
				status: failure.startsWith('unknown') ? 'unknown' : 'success'
			};
		},
		async verifyRemove(target, snapshot) {
			metrics.verifyRemove++;
			await wait('verifyRemove', snapshot.context.scope, snapshot.operationId, target.key);
			return {
				key: target.key,
				status: controls.verifyUnknown
					? 'unknown'
					: deletions.get(
								identity(snapshot.context.scope, snapshot.operationId, target.key)
						  )
						? 'success'
						: 'unsent'
			};
		}
	};
	async function detail(key: string | number, context: AsamRequestContext) {
		metrics.load++;
		await wait('detail', context.scope, undefined, key);
		if (controls.loadError) throw new Error(text('受控详情加载失败'));
		if (controls.forbidden)
			return { status: 'forbidden' as const, message: text('服务拒绝读取详情') };
		const row = store(context.scope).get(key);
		return controls.missing || !row
			? { status: 'missing' as const }
			: { status: 'ready' as const, data: { ...row } };
	}
	return { services, detail, controls, metrics };
}

		verify: 0,
		verifyRemove: 0,
		events: []
	});

	const writes = new Map();
	const deletions = new Map();

		if (!stores.has(scope))
			stores.set(
				scope,

					Array.from({ length: 100 }, (_, i) => {
						const id = i === 1 ? '0' : i;

		const event = { name, scope, operationId, key, start: performance.now(), end: 0 };
		await new Promise((resolve) => setTimeout(resolve, controls.delay));
		event.end = performance.now();

		const failure = controls.save;

		if (failure === 'field-error')
			return {
				status: 'failure',
				message: text('服务字段校验失败'),

			};

			return { status: 'failure', message: text('服务明确拒绝，未写入') };

			await wait('list', context.scope);

				Object.entries(request.query).every(

			if (request.sort) {
				const { field, order } = request.sort;
				rows.sort(
					(a, b) =>
						String(Reflect.get(a, field)).localeCompare(
							String(Reflect.get(b, field)),
							'zh-CN',
							{ numeric: true }
						) * (order === 'ascending' ? 1 : -1)
				);
			}
			return {
				list: rows
					.slice(
						(request.currentPage - 1) * request.pageSize,
						request.currentPage * request.pageSize
					)
					.map((row) => ({ ...row, name: row.name.replace('资产', controls.prefix) })),
				total: rows.length
			};
		},
		async defaults(context) {
			metrics.defaults++;
			await wait('defaults', context.scope);
			return { name: '', team: '研发', status: '启用', owner: '成员0', email: '', note: '' };
		},
		async load(key, context) {
			metrics.load++;
			await wait('load', context.scope, undefined, key);
			if (controls.loadError) throw new Error(text('受控编辑加载失败'));
			const row = store(context.scope).get(key);
			if (!row) throw new Error(text('记录已不存在'));
			return { ...row };
		},
		create(model, operation, context) {
			metrics.create++;
			return save(model, context.scope, operation.target.operationId);
		},
		update(key, model, operation, context) {
			metrics.update++;
			return save(model, context.scope, operation.target.operationId, key);
		},
		async verify(operation, context) {
			metrics.verify++;
			await wait('verify', operation.scope, operation.target.operationId);
			if (controls.verifyUnknown) return { status: 'unknown' };
			return writes.get(identity(operation.scope, operation.target.operationId))
				? { status: 'success', data: { verified: true } }
				: { status: 'not-applied' };
		},
		async resolve(keys, context) {
			return keys.map((key) => {
				const row = store(context.scope).get(key);
				return row
					? {
							key,
							target: {
								key,
								version: String(row.version),
								payload: { name: row.name }
							}
						}
					: { key, reason: text('记录不存在') };
			});
		},
		async remove(target, snapshot) {
			metrics.remove++;
			const failure = controls.partial && target.key === '0' ? 'failure' : controls.remove;
			await wait('remove', snapshot.context.scope, snapshot.operationId, target.key);
			if (failure === 'failure' || failure === 'field-error')
				return { key: target.key, status: 'failure', message: text('明确未应用，可重试') };
			const applied = failure !== 'unknown-not-applied';
			if (applied) store(snapshot.context.scope).delete(target.key);
			deletions.set(
				identity(snapshot.context.scope, snapshot.operationId, target.key),
				applied
			);
			if (controls.refreshError) controls.listError = true;
			return {
				key: target.key,
				status: failure.startsWith('unknown') ? 'unknown' : 'success'
			};
		},
		async verifyRemove(target, snapshot) {
			metrics.verifyRemove++;
			await wait('verifyRemove', snapshot.context.scope, snapshot.operationId, target.key);
			return {
				key: target.key,
				status: controls.verifyUnknown
					? 'unknown'
					: deletions.get(
								identity(snapshot.context.scope, snapshot.operationId, target.key)
						  )
						? 'success'
						: 'unsent'
			};
		}
	};
	async function detail(key, context) {
		metrics.load++;
		await wait('detail', context.scope, undefined, key);
		if (controls.loadError) throw new Error(text('受控详情加载失败'));
		if (controls.forbidden) return { status: 'forbidden', message: text('服务拒绝读取详情') };
		const row = store(context.scope).get(key);
		return controls.missing || !row
			? { status: 'missing' }
			: { status: 'ready', data: { ...row } };
	}
	return { services, detail, controls, metrics };
}

使用说明

从 @asam-ui/pro 导入 useAsamCrudController 与 AsamCrudPage，组合查询、列表、编辑和详情。本页示例服务使用合成数据，真实业务服务由应用注入。

标准模式默认将查询与表格分卡显示，查询支持展开/收起；提交查询或重置回到第一页，工具栏刷新重新读取当前查询并保留当前页，期间有忙碌与完成反馈。空表格使用公共空状态，内容区至少 400px，并提供重新加载；自定义空态插槽仍按原约定使用。

tableSize 控制表格实际行高：large/default/small/mini 分别为 60/50/42/36px，默认 default（50px）。它与 size 控制的输入框、按钮等控件尺寸独立，密度菜单不将控件高度当作行高。标准行操作默认 small，保持单行，超过三项或空间不足时进入真实更多菜单。

标准详情默认从名称或首个可见业务列打开右侧抽屉，桌面最小 800px、窄屏受视口限制，并开启原生拖动调宽；关闭后回到真实触发元素。detailEntry 可指定唯一详情列或显式关闭，detailOnRowDblclick 可独立开启行双击，交互控件不触发行双击。detailDisplay="dialog" 和完整 detail 替换槽仍可使用；也可单独使用 AsamDetailDialog 或 AsamDetailDrawer 展示同一详情控制器的资料，或用 AsamDetailPanel 将资料平铺在页面内。

编辑表单沿用公共分组、字段和校验契约：分组标题加粗；非 inline 校验消息使用警告色并绝对定位，不因消息出现推动字段。原生 inlineMessage 及自定义错误插槽保持原行为。Schema、Dialog、Drawer 和 Steps 的汇总入口占固定区域，支持展开列表与 Escape 关闭；Schema、Dialog 和 Drawer 的问题项支持点击定位，Steps 列表展示完整问题文字，不截断消息或改变校验结果。

业务状态用真实 AsamTag 呈现，由应用按 typed 原值配置：进行中用 primary；成功、完成、启用、已发布、已审核等用 success；等待或未确定用 warning；错误、拒绝、驳回用 danger；中性状态用 info。相同文案的不同来源不自动覆盖映射，类别、平台等枚举不冒充状态。通用 CRUD 不内置芋道业务字典。

用于同页查询、分页表格、详情和新增/编辑弹窗、单行/明确所选目标删除。仅需要读取时使用 useAsamListController；仅编辑使用 useAsamEditor；不需要页级协调时不增加CRUD层。此模板不拥有业务URL、鉴权系统、路由或后端幂等性。

事实源与服务边界

列表控制器的ListController拥有查询、排序、分页、原Row与结果；编辑与详情协议的Editor/WriteCoordinator拥有模型转换、校验、稳定operationId与unknown核实，Detail拥有互斥direct/load详情；表格操作的Selection/OperationCoordinator拥有typed key跨页选择、版本/确认快照、逐目标记录与列偏好。CRUD层只记录成功结果的刷新归属，不能赋值page.rows/model或建立第二套分页。

宿主持有runtime、writes、operations；可传入既有list，页面不dispose这些借入资源。未传list时在自身effectScope创建并释放；页面拥有的editor/detail/selection/preferences随controller.dispose或创建scope退出释放。AsamCrudPage卸载只释放视图注册和焦点；宿主自行决定controller范围，跨路由未决记录保留到核实，不承诺进程退出后持久化。

Row/Query/Edit/Key泛型明确；Key仅string或有限number，0与字符串0独立。queryCodec/editCodec通过encode/decode显式校验业务模型；字段Record不会被直接断言成业务类型。提供list/defaults/load/create/update/verify/resolve/remove/verifyRemove全部服务，签名见Types.ts。列表Row不直接作为可变编辑模型，load返回经字段适配和克隆后的模型。

写入与删除

operationId由宿主生成。create/edit可显式传ID；同一目标未决时重开优先沿用原ID，编辑与详情协议拒绝重复派发，已成功结果只交付一次。取消确认不发请求；发送后异常记unknown，关闭/取消不是服务未生效。verify明确success/not-applied/unknown，只有明确not-applied才能显式再保存。服务failure必须明确证明未写入。

editCodec.decode在编辑与详情协议公开validatePayload阶段执行一次，随后核宿主业务校验与当前editor/list会话、scope/epoch和禁用/只读状态；仅以本次临时转换输入交付保存桥，不维护另一套写入事实。服务前再复核当前资格及宿主权限/键判定后的context；确未调用服务的晚拒绝返回failure，由编辑与详情协议原协调器记not-applied，不用throw伪装unknown。已合法进入服务后的异常仍是unknown。未派发或明确未应用的同ID重新打开可绑定当前来源，真实saving/unknown/未交付成功的原归属继续保留。

删除resolve为每个typed key返回带version/payload的目标或拒绝原因，confirm接收准确合法目标与拒绝列表。每次发送前由表格操作再解析版本及权限。删除按success/failure/unknown/unsent逐目标记录；failure必须明确未应用，unknown阻止新ID绕过，核实可返回success/unsent/unknown；retry只续failure/unsent，既成功目标不重发。一个unknown会停止后续目标。失败、取消、部分成功保留原选择和可核记录；全部成功且原范围仍有效时清空选择，焦点归页标题。部分成功后用记录的“重试未应用项”，不要为同一批目标换ID。

写成功后原scope/epoch/session/查询/排序/页码仍有效才发一次list.refresh。刷新失败单独显示并允许retryRefresh；保存/删除成功不会倒写为失败。查询或范围变化后的旧结果只留写记录，不刷新新范围或清空新选择。删除当前页最后数据后，使用既有分页controller的一次校正链，不维护第二个页码。核实或重试产生新增成功目标时仅对新增部分触发一次刷新。权限在打开后、确认后及发送前均重核；已合法发出的记录继续保留。

状态与可访问性

queryCodec接收列表控制器转换后的查询，空查询字段可能省略；示例补空条件后仍校验实际类型。列表控制器行动回调交付行快照，页面按typed key重取当前Row事实后才打开编辑/详情。重试删除的确认仅展示本轮failure/unsent目标，成功目标不会再次列为将删除对象；列表刷新期间重试入口禁用。页面只展示、核实本数据集的CRUD记录，宿主原协调器的其他行动仍由宿主直接管理。

readonly允许查询/翻页/详情和结果核实，禁止页面新增/编辑/删除及写回调；disabled阻断本页新交互，已有合法在途结果按协议归属处理。宿主直接使用其原list/runtime不被页面拦截。控件四档large/big/default/small对应40/36/32/28，继承ASAM tokens；详情字段为只读展示。390窄屏表格在内部横向滚动，表单响应式，键盘可达确认取消、错误汇总和字段定位；动作反馈不延迟交互，遵守减少动态。

自有list首次读取由页面调度：初始禁用、排队窗口或初始化异步校验期间禁用均不调用list服务；合法启用后只恢复尚未派发的初始化一次，之后切换disabled不自动重读。发送边界取消沿用列表控制器的公开page.cancel协议，不注入空成功结果或伪造服务错误；已合法调用的请求不因disabled变化撤销。借入宿主list不建立此初始化调度，也不拦截其直接请求；readonly不阻止合法读取。

queryCodec属于宿主公开适配器，其同步副作用可能改变页面disabled、销毁状态或runtime scope。服务桥在完成解码后、真正调用list服务前再次核当前资格与context/session；过期身份不能进入服务，也不会因其过期而取消页面当前的另一读取。codec自身已发生的宿主副作用不会被撤销，已合法进入服务后的副作用仍按原请求协议处理。

示例默认 100 条记录、每页 20 条、4 个查询与 6 个编辑/详情字段；这些是示例规模，协议不固定字段数量。真实业务的数据量、服务耗时与设备性能应单独评估。
