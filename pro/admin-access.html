import { scopeKey } from '@asam-ui/pro/application/security';
import type { AdminHost } from './Types.js';
export function adminView(host: AdminHost) {
	const decision = host.security.check(host.pageRule);
	const state = host.security.state.value;
	if (decision === 'pending') return { status: 'pending' as const, entries: [], records: [] };
	if (decision !== 'allowed' || !state.subject)
		return { status: 'denied' as const, entries: [], records: [] };
	const value = host.projection.value;
	if (!value || value.owner !== scopeKey(state.subject) || value.epoch !== state.epoch)
		return { status: 'unavailable' as const, entries: [], records: [] };
	return {
		status: 'ready' as const,
		entries: value.entries.filter((item) => host.security.check(item.rule) === 'allowed'),
		records: value.records.filter((item) => host.security.check(item.rule) === 'allowed')
	};
}
export function adminTarget(host: AdminHost, id: string): string | null {
	const view = adminView(host);
	return view.status === 'ready'
		? (view.entries.find((item) => item.id === id)?.path ?? null)
		: null;
}

	};

复用原createSecurity与createSecurityNavigation。入口及记录分别核权限，点击时重查；资料绑定当前主体、工作空间和epoch。刷新或失效立即撤下旧内容，直接地址与历史返回经过原路由守卫。

页面接入

AdminHost提供只读资料和原权限控制器，页面发送入口标识，由宿主完成导航。演示使用实际MemoryRouter和只读入口摘要，不提供成员或角色增删改。支持中英文、暗色、四档尺寸及窄屏。

接入边界

演示身份和记录全部为本地合成，不连接真实登录或管理服务。正常发布包、源码展示、复制、ZIP和运行同源。替换host应重新挂载。JS checkJs=false不代表JS脚本类型检查。
