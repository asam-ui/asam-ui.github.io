function replaceReceipt(status: RegistrationReceipt['status']) {
	receipt.value = {
		id: `demo-${++revision}`,
		email:
			status === 'expired'
				? 'linlan@registration-confirmation-for-workspace-members.example.invalid'
				: 'linlan@example.invalid',
		status
	};
	destination.value = undefined;
}
function setLocale(value: unknown) {
	if (value === 'en' || value === 'zh-CN') props.onLocale?.(value);
}
function setSize(value: unknown) {
	if (sizes.some((item) => item === value)) size.value = value as (typeof sizes)[number];
}
function setMode(value: unknown) {
	if (value === 'success' || value === 'failure' || value === 'late') mode.value = value;
}
const resend: RegistrationResultProps['resend'] = async (_receipt, request) => {
	const selected = mode.value;
	await new Promise<void>((resolve, reject) => {
		const abort = () => {
			clearTimeout(timer);
			reject(new DOMException('Aborted', 'AbortError'));
		};
		const timer = setTimeout(
			() => {
				request.signal.removeEventListener('abort', abort);
				resolve();
			},
			selected === 'late' ? 2500 : 250
		);
		request.signal.addEventListener('abort', abort, { once: true });
		if (request.signal.aborted) abort();
	});
	if (!request.isCurrent()) throw new DOMException('Aborted', 'AbortError');
	if (selected === 'failure') throw Error('本地合成邮件服务失败');
	return { status: 'sent', retryAfter: 8 };
};

<script setup lang="ts">
import { computed, onScopeDispose, shallowRef, watch } from 'vue';
import {
	AsamButton,
	AsamCard,
	AsamDescriptions,
	AsamDescriptionsItem,
	AsamResult
} from '@asam-ui/ui';
import type { RegistrationResultEmits, RegistrationResultProps } from './Types.js';
import { registrationResultText, useRegistrationResultI18n } from './I18n.js';
import type { RegistrationResultLabel } from './I18n.js';

const props = defineProps<RegistrationResultProps>();
const emit = defineEmits<RegistrationResultEmits>();
const i18n = useRegistrationResultI18n();
const t = (key: RegistrationResultLabel) => registrationResultText(i18n, key);
const expired = shallowRef(props.receipt.status === 'expired');
const verified = computed(() => props.receipt.status === 'verified');
const displayEmail = computed(() => {
	const email = props.receipt.email;
	const separator = email.lastIndexOf('@');
	const account = email.slice(0, separator);
	if (separator <= 0 || separator === email.length - 1) return '—';
	if (account.endsWith('***')) return email;
	return `${account.slice(0, Math.min(2, account.length - 1))}***${email.slice(separator)}`;
});
const pending = shallowRef(false);
const remaining = shallowRef(0);
const feedback = shallowRef<'sent' | 'failure' | 'expired'>();
let controller: AbortController | undefined;
let timer: ReturnType<typeof setInterval> | undefined;
let active = true;

function invalidate() {
	controller?.abort();
	controller = undefined;
	clearInterval(timer);
	timer = undefined;
	pending.value = false;
	remaining.value = 0;
	feedback.value = undefined;
}
watch(
	() => [props.receipt.id, props.receipt.email, props.receipt.status],
	() => {
		invalidate();
		expired.value = props.receipt.status === 'expired';
	},
	{ flush: 'sync' }
);
watch(
	() => [props.disabled, props.readonly],
	() => {
		if (pending.value) invalidate();
	},
	{ flush: 'sync' }
);

async function send() {
	if (
		!active ||
		verified.value ||
		props.disabled ||
		props.readonly ||
		pending.value ||
		remaining.value
	)
		return;
	const own = new AbortController();
	controller = own;
	const receipt = Object.freeze({ ...props.receipt });
	const isCurrent = () =>
		active &&
		controller === own &&
		!own.signal.aborted &&
		props.receipt.id === receipt.id &&
		props.receipt.email === receipt.email &&
		props.receipt.status === receipt.status &&
		!props.disabled &&
		!props.readonly;
	pending.value = true;
	feedback.value = undefined;
	try {
		const result = await props.resend(receipt, { signal: own.signal, isCurrent });
		if (!isCurrent()) return;
		expired.value = result === 'expired';
		feedback.value = result === 'expired' ? 'expired' : 'sent';
		if (result !== 'expired') {
			const retryAfter = result === 'sent' ? 60 : result.retryAfter;
			if (!Number.isSafeInteger(retryAfter * 1000) || retryAfter <= 0) return;
			const until = Date.now() + retryAfter * 1000;
			remaining.value = Math.ceil(retryAfter);
			timer = setInterval(() => {
				remaining.value = Math.max(0, Math.ceil((until - Date.now()) / 1000));
				if (!remaining.value) {
					clearInterval(timer);
					timer = undefined;
				}
			}, 250);
		}
	} catch {
		if (isCurrent()) feedback.value = 'failure';
	} finally {
		if (controller === own) {
			controller = undefined;
			pending.value = false;
		}
	}
}
onScopeDispose(() => {
	active = false;
	invalidate();
});
</script>

	const account = email.slice(0, separator);
	if (separator <= 0 || separator === email.length - 1) return '—';
	if (account.endsWith('***')) return email;
	return `${account.slice(0, Math.min(2, account.length - 1))}***${email.slice(separator)}`;

	controller?.abort();

	clearInterval(timer);
	timer = undefined;
	pending.value = false;
	remaining.value = 0;
	feedback.value = undefined;

		props.disabled ||
		props.readonly ||

		remaining.value

回执变化、禁用、只读及卸载会取消当前请求；迟到响应不会覆盖新回执。独立工程的 App.vue 仅为合成演示宿主；mountRegistrationResult 返回幂等销毁函数，不销毁外部 i18n。仅已有外层ASAM主题时开启 inheritAppearance。
