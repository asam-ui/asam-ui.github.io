<script setup lang="ts">
import { computed, inject, onBeforeUnmount, onMounted, ref } from 'vue';
import { AsamButton, AsamInputNumber, defineAsamControl, defineAsamOptions } from '@asam-ui/ui';
import { AsamSearchForm, useAsamListController } from '@asam-ui/pro/components';
import type { AsamFieldDefinition, AsamFieldModel } from '@asam-ui/pro/components';
import { createLatestRequest, RequestFailure } from '@asam-ui/pro/application/transport';
import { useScaffoldI18n } from './I18n.js';
import { contextKey } from './context.js';
import { decodePage } from './Mock.js';
import type { Page, Scenario, SyntheticRecord } from './Mock.js';
import settings from './settings.json';
const context = inject(contextKey);
if (!context) throw new Error('Host context required');
const { preferences, client, mock } = context;
const i18n = useScaffoldI18n();
const result = ref<Page>({ rows: [], total: 0, page: 1, pageSize: 20 });
const scenarios: Scenario[] = ['normal', 'empty', 'business', 'network', 'offline', 'late'];
const resetValues = { query: '', seed: settings.seed, scenario: 'normal' };
const seedControl = defineAsamControl<unknown, object>({
	component: AsamInputNumber,
	emptyValue: () => null,
	sourceKinds: []
});
const fields = computed<readonly AsamFieldDefinition[]>(() => [
	{
		key: 'query',
		kind: 'text',
		label: i18n.t('app.templates.scaffold.query')
	},
	{
		key: 'seed',
		kind: 'custom',
		label: i18n.t('app.templates.scaffold.seed'),
		control: seedControl,
		isValue: (value) => value == null || typeof value === 'number',
		emptyValue: () => null,
		isEmpty: (value) => value == null,
		query: (value) => value,
		submit: (value) => value,
		format: (value) => String(value ?? ''),
		modes: { query: { componentProps: { min: 0, max: 4294967295 } } }
	},
	{
		key: 'scenario',
		kind: 'single',
		label: i18n.t('app.templates.scaffold.scenario'),
		source: defineAsamOptions().static({
			options: scenarios.map((value) => ({ value, label: value }))
		})
	}
]);
const errorCode = ref<'' | 'cancelled' | 'request' | 'invalidData' | 'invalidSeed'>('');
const failedKind = ref('');
const error = computed(() => {
	switch (errorCode.value) {
		case 'cancelled':
			return i18n.t('app.templates.scaffold.cancelled');
		case 'request':
			return i18n.t('app.templates.scaffold.requestFailed', { kind: failedKind.value });
		case 'invalidData':
			return i18n.t('app.templates.scaffold.invalidData');
		case 'invalidSeed':
			return i18n.t('app.templates.scaffold.invalidSeed');
		default:
			return '';
	}
});
const loading = ref(false);
let revision = 0;
const latest = createLatestRequest<Page>(client, (value) => {
	result.value = value;
});
const identity = (id: SyntheticRecord['id']) =>
	`${id === null ? 'null' : typeof id}:${JSON.stringify(id)}`;
// 生成工程使用固定版本的 Pro 包，查询界面接入该版本已有的控制器契约。
const queryController = useAsamListController<SyntheticRecord>({
	fields,
	initial: { ...resetValues, scenario: settings.scenario },
	immediate: false,
	rowKey: (row) => identity(row.id),
	async api({ query, currentPage }, { signal }) {
		try {
			if (typeof query.seed !== 'number') throw new TypeError('Expected a numeric seed');
			mock.reset(query.seed);
		} catch (error) {
			errorCode.value = 'invalidSeed';
			throw error;
		}
		const data = await requestPage(currentPage, query, signal);
		return { list: data.rows, total: data.total };
	}
});
queryController.setResetBaseline(resetValues);
async function requestPage(page: number, filters: AsamFieldModel, signal?: AbortSignal) {
	const current = ++revision;
	errorCode.value = '';
	loading.value = true;
	try {
		mock.setScenario(scenarios.find((value) => value === filters.scenario) ?? 'normal');
		return await latest.load(
			{
				url: '/synthetic/records',
				params: { query: String(filters.query ?? ''), page, pageSize: 20 }
			},
			{ decode: decodePage, signal }
		);
	} catch (cause) {
		if (current === revision) {
			errorCode.value =
				cause instanceof RequestFailure
					? cause.kind === 'cancelled'
						? 'cancelled'
						: 'request'
					: 'invalidData';
			failedKind.value = cause instanceof RequestFailure ? cause.kind : '';
		}
		throw cause;
	} finally {
		if (current === revision) loading.value = false;
	}
}
function load(page = 1) {
	void requestPage(page, queryController.draft.value).catch(() => {
		// 请求失败已通过页面状态呈现。
	});
}
function cancel() {
	latest.cancel();
}
function recover() {
	queryController.setDraft({ ...queryController.draft.value, scenario: 'normal' });
	void load();
}
onMounted(() => {
	void load();
});
onBeforeUnmount(() => {
	revision++;
	latest.dispose();
});
</script>

export type Scenario = 'normal' | 'empty' | 'business' | 'network' | 'offline' | 'late';
export interface SyntheticRecord {
	id: number | string | null | false;
	name: string;
	amount: number;
	createdAt: string;
}
export interface Page {
	rows: SyntheticRecord[];
	total: number;
	page: number;

	seed: number;
	scenario?: Scenario;
	delayMs?: number;
	count?: number;
	/** Injection makes delay/cancellation tests independent of wall time. */
	schedule?: (callback: () => void, delay: number) => () => void;
}

const timeBase = 1704067200000;
export function records(seed: number, count = 100): SyntheticRecord[] {
	if (!Number.isInteger(seed) || seed < 0 || seed > 0xffffffff)
		throw new Error('uint32 seed required');
	if (!Number.isInteger(count) || count < 0 || count > 10000)
		throw new Error('count must be 0..10000');
	let state = seed >>> 0;
	const next = () => {
		state = (Math.imul(state, 1664525) + 1013904223) >>> 0;
		return state;
	};
	const identities: SyntheticRecord['id'][] = [0, '0', null, false];
	return Array.from({ length: count }, (_, index) => ({
		id: index < 4 ? identities[index]! : index,
		name: `合成记录 ${index + 1} · ${next().toString(16).slice(0, 6)}`,
		amount: next() % 10000,
		createdAt: new Date(timeBase + (next() % 365) * 86400000).toISOString()

	let seed = options.seed;
	let data = records(seed, options.count);
	let scenario = options.scenario ?? 'normal';
	let disposed = false;
	let sequence = 0;
	const pending = new Map<number, () => void>();
	const schedule =
		options.schedule ??
		((callback, delay) => {
			const timer = setTimeout(callback, delay);
			return () => clearTimeout(timer);
		});
	function setScenario(next: Scenario) {
		if (!scenarios.includes(next)) throw new Error('Unknown scenario');
		scenario = next;
	}
	setScenario(scenario);
	const delay = options.delayMs ?? 160;
	if (!Number.isInteger(delay) || delay < 0 || delay > 30000)
		throw new Error('delayMs must be 0..30000');

		const mode = scenario;
		const snapshot = data;
		const number = ++sequence;
		return new Promise((resolve, reject) => {
			let cancelTimer = () => {};
			let settled = false;
			const cleanup = () => {
				cancelTimer();
				context.signal.removeEventListener('abort', abort);
				pending.delete(number);
			};
			const abort = () => {
				if (settled) return;
				settled = true;
				cleanup();
				reject(new TransportFault('cancelled'));
			};
			if (disposed || context.signal.aborted) {
				abort();
				return;
			}
			context.signal.addEventListener('abort', abort, { once: true });
			pending.set(number, abort);
			cancelTimer = schedule(

					if (settled) return;
					settled = true;

						return;

					const pageSize = Number(input.params?.pageSize ?? 20);
					if (
						input.url !== '/synthetic/records' ||
						!Number.isInteger(page) ||
						page < 1 ||

接口与行为

入口 Generator.mjs，声明 Generator.d.mts / Types.d.ts。Node >=22.19.0 <23、pnpm 11.9.0；生成时需要完整固定 Git 来源，生成后的工程不依赖本仓库。

name asam-demo 小写 npm 包名，支持 scope，最多214字符；new模式可重复改名，integrate保留宿主名，显式不同name报冲突

调用 返回 / 错误 契约
plan(options) ScaffoldPlan；无效输入抛错 不写目标；文件 add/modify/unchanged/conflict、原/新hash、原bytes、来源、package结构差分及锁/配置guard；CLI dry-run 同义
apply(plan) applied / unchanged / conflict / partial 输入 seal 与来源核验，目标逐路径核链接和指纹；冲突不写；同参数无变化不碰manifest
verify(options) verified / changed 对manifest所有自有文件逐hash核验，缺失为null；不安装、不构建、不修复
apply(plan,{failAfter}) partial 故障注入仅用于恢复测试；不是业务选项或强制覆盖

所有入口同步执行，错误为 Error。CLI 无效输入退出1；conflict/partial/changed退出2；成功退出0。CLI 可将输出写到显式尚不存在的文件，否则stdout输出JSON。无每次生成审批。

manifest：.asam/scaffold/<namespace>.json，包含稳定文件ID、同版source、bytes/hash和完整原始base64。参数变化只更新hash仍等生成器原件的自有文件；用户改动/删除保留并报告冲突。未拥有的同名文件不覆盖。package采用JSON结构合并，保留无关依赖/脚本；既有.npmrc只记录guard哈希、保留宿主原文并诊断冲突，不把凭据原文复制到新plan或ownership manifest；pnpm11的严格peer、engine、linker/copy等有效设置写入工程自己的workspace YAML。既有workspace YAML仅支持与模板目标相同的配置，复杂workspace需宿主单独迁移，不能自动重写。

.pending.json为恢复日志。写入前持久化整批inFlightBatch意图；部分失败保留已完成路径及原计划，原样重新apply。按每项after哈希识别已写文件，恢复时先补刷盘、再持久化已恢复完成记录，不重复写字节；兼容首版inFlight单项日志。已完成字节变化则拒绝，禁止删除整个输出后重跑或自动撤销用户内容。manifest已完成后中断可清除本plan日志，但仍核全部自有文件与未拥有的guards。底层文件系统在单文件write中发生损坏时不猜测回滚，保留故障字节并人工核原plan；整批不宣称原子事务。

完成核验点在数据文件和完成日志持久化之后、写完成manifest之前：逐项重新读取全部自有文件（含本轮unchanged项）以及包、锁、配置等guards。发现变化返回partial，changed?: {path,expected,actual}[]列具体指纹差异，用户字节及已有manifest不被回退或替换。此为明确的顺序核验边界；之后的外部编辑由verify诊断，不承诺永久无变化，也不锁住编辑器。guard同时防止plan后编辑；宿主为每个输出目录安排唯一写者。

公共API保持同步。内部Durability.mjs用Node内置worker批量等待独立文件的真实fsync；首个数据文件、意图/完成日志与manifest仍同步刷盘，其他数据文件逐项fsync完成后才进入最终核验。写句柄保留至刷盘屏障之后再关闭，避免串行关闭脏文件；worker使用独立打开的句柄，仅刷盘和关闭，不写应用字节。错误或30秒看门狗超时只返回partial，保留恢复计划；没有略过持久化、减小默认数据量或扣除生成工作计时。

Mock API在生成应用 src/<ns>/Mock.ts：records(seed,count=100)固定合成时基2024-01-01和32位LCG；createMock({seed,scenario='normal',delayMs=160,count=100,schedule?})返回请求模板协议transport、setScenario、reset、pending、dispose。count范围0～10000；默认示例100条、每页20条；更大规模需另评估。传输参数url固定/synthetic/records，params为page正整数、pageSize 1～100、query字符串。data分支返回{rows,total,page,pageSize}，decodePage(unknown)验证一次。id保留0、字符串0、null、false。business/network/offline为真实transport拒绝；late为8倍延迟；取消通过AbortSignal清理定时器；reset取消在途并恢复种子；dispose不再启用。schedule可注入手动时钟复现相反完成顺序。请求模板 Latest负责旧代保护，不另建请求池。

mountApplication(element)异步返回幂等disposer，实例拥有Router/Navigation/Preferences/Mock/Client，卸载依次释放。资源仅data/about显式allowlist；偏好owner为['synthetic-demo',0]并使用namespace作app。多实例应各自传入独立element与不同应用namespace；无全局controller。包、样式仅正常公开入口，vendor是固定Git源码快照并列来源hash，修改vendor将被后续生成报告为用户冲突。

离线工具Offline.mjs入口：prepareOffline({project,kit,store})只读实际安装与有效锁、pnpm11.9.0 v11缓存index，输出新材料目录。保留完整锁中的版本/integrity及已装平台的原index BLOB、逐SHA512复核的CAS内容、正常双包/EP补丁；其他平台可选依赖记录在锁中但不冒已准备。checkOffline({project,kit})先核同锁/平台/材料hash，缺项抛错且无网络；restoreOffline({...})核验后仅向本机统一store补缺，已有异字节拒绝，不覆盖现有cache对象、不改全局配置。Windows仅D:/.pnpm-store/v11；其他宿主用其统一v11目录，当前材料格式仅支持 win32-x64。SQLite采用事务及busy timeout，现有index行只保留，不替换。此格式绑定已核pnpm11.9.0实现，升级pnpm应另适配验证。

CLI为node Offline.mjs prepare|check|restore offline-options.json，options包含project、全新kit和统一store绝对路径。工具从不发网络请求或自动安装；恢复准备完成后再在project运行pnpm install --offline --frozen-lockfile，然后typecheck/build/preview。生成工程自己的workspace YAML承载补丁/overrides/allowBuilds，不能传--ignore-workspace绕过其配置。准备目录是不可变外置材料，不应设置成consumer的另一个store。

同版脚手架自有文本资产输出统一LF；依赖快照则逐raw Git blob保留原bytes，不经过working-tree EOL或export filter。目标中用户自行更改行尾仍按原始hash识别为dirty，不自动归一化覆盖。

manifest 的 source 与 SOURCES.json 的 templateSource 标明依赖模板来源；生成器自身版本和资料身份分别核对，不把模板来源当作生成器版本。

使用说明

生成完整可编辑 Vue 应用，组装导航、偏好与请求模板。示例使用合成数据，真实业务后端由应用实现。

支持单工程 ESM Vue 3.5.34 / Vite 8.0.10 / TS 6.0.3，新建与兼容工程独立命名空间接入。Node >=22.19.0 <23、pnpm11.9.0、plugin-vue6.0.9、vue-tsc3.3.11、Vue Router4.6.3、EP2.14.7固定，不改版本迁就本机。React、Nuxt、旧Vue、复杂workspace自动迁移不支持。兼容接入新增独立HTML入口和命名空间脚本，宿主原HTML、main、依赖、脚本保留；不自动导入宿主路由。

json
{
	"root": "/absolute/existing/workspace",
	"output": "演示工程 with spaces",
	"mode": "new",
	"name": "@example/asam-demo",
	"namespace": "asam",
	"title": "合成数据演示",
	"locale": "zh-CN",
	"theme": "light",
	"seed": 20261018,
	"scenario": "normal",
	"packages": {
		"ui": "/absolute/path/to/ui.tgz",
		"pro": "/absolute/path/to/pro.tgz"
	}
}

根目录必须先存在。保存为options.json，再执行：

text
node templates/scaffold/cli.mjs plan options.json plan.json
node templates/scaffold/cli.mjs apply plan.json
node templates/scaffold/cli.mjs verify options.json

API等价入口见API.md。plan文件输出必须新建，方便保留真实输入；参数变更另存新plan。审阅结构化差分后可无人值守apply，无内置审批。不要编辑plan内容；修改参数应重做plan。apply返回partial时仅重试原plan，检查精确冲突再人工迁移，不force、不删除整包、不自动回退用户修改。

apply在数据和恢复日志刷盘后、写完成manifest前，核全部自有文件与包/锁/配置guards。中途修改已经写过或本轮unchanged的文件也返回partial，changed字段列路径与指纹，保留用户内容和旧manifest。完成核验点之后的外部编辑由verify报告。批量持久化意图支持崩溃恢复，兼容旧inFlight日志；恢复核对原计划并补刷盘，不重复已完成文件写入。生成API仍同步，内部worker不需要额外依赖；生成的Vue应用不依赖worker或原仓库。

在生成工程cwd执行（Windows每个进程先加载正确存储入口）：

sh
pnpm install --frozen-lockfile
pnpm run asam:typecheck
pnpm run asam:build
pnpm run asam:dev

命名空间非asam时替换脚本前缀。访问/asam.html，使用真实导航/页签、语言、亮暗、四尺寸和合成分页查询。数据页保留身份类型0/字符串0/null/false；切empty/business/network/offline/late再查询，可观察真实请求时序及恢复。所有数据和菜单明确合成，禁止替换为推测的真实CL API或账号。亮暗采用Provider和ASAM tokens；尺寸large/big/default/small映射40/36/32/28；减少动态系统设置关闭入场动画。

可编辑DataPage.vue字段映射/列表、settings.json种子/默认场景、application.ts导航与偏好owner。修改后类型检查/生产构建/浏览器观察；若需再次生成，manifest会准确报告这些用户修改，不擅自覆盖。移动整个生成工程（含.asam本地正常包、源码、配置及有效锁）后，在新路径自行安装依赖运行；依赖在目标工程中重新安装，不复制其他工程的 node_modules。

生成器要求其支持材料中规定的 UI/Pro 普通包，逐字节核对后使用；任意同名或同版本包不能替代。依赖模板快照及来源记录随生成器材料保留，接入前核对实际支持的材料。

EP类型补丁从固定Git提取；两条tiptap3.31.3 overrides与严格peer策略同源。普通工程verifyDepsBeforeRun=error，差异应修真实配置，不反复强制重装。生成器不安装依赖、不写机器全局配置、不写根workspace或根锁。

离线能力分开：已有同版材料支持pnpm install --offline --frozen-lockfile及构建；已有dist支持断外网本地preview；缺依赖的在线准备是单独动作，不宣称可离线完成。外置材料准备/核验/恢复见Offline.mjs配套API，完整版本/integrity与正常包、补丁、锁随材料保留；始终恢复到本机统一store，不能为临时工程另建共享 store。缺材料明确拒绝，不静默联网。浏览器预览仅本地资源，不含在线字体、CDN、真实服务或离线模型。

默认示例 100 条记录、每页 20 条。更大数据量、不同设备和实际业务服务应单独评估；离线运行要求预先准备完整依赖。

Types.d.ts · 完整类型
ts
export type Scenario = 'normal' | 'empty' | 'business' | 'network' | 'offline' | 'late';
export interface ScaffoldOptions {
	/** Existing absolute directory; generator never writes outside root/output. */
	root: string;
	/** Explicit safe relative path; spaces and Unicode supported. */
	output: string;
	mode?: 'new' | 'integrate';
	name?: string;
	title?: string;
	namespace?: string;
	locale?: 'zh-CN' | 'en-US';
	theme?: 'light' | 'dark' | 'system';
	seed?: number;
	scenario?: Scenario;
	/** Immutable normal producer packages, required on first generation. */
	packages?: { ui: string; pro: string };
}
export interface PlanEntry {
	id: string;
	path: string;
	action: 'add' | 'modify' | 'unchanged' | 'conflict';
	reason?: string;
	before: string | null;
	after: string;
	bytes: number;
	content: string;
	source: string;
}
export interface ScaffoldPlan {
	id: string;
	version: string;
	source: string;
	target: string;
	namespace: string;
	manifestPath: string;
	pendingPath: string;
	guards: Record<string, string | null>;
	entries: PlanEntry[];
	packageDiff: { before: string | null; after: string };
}
export interface ApplyResult {
	status: 'applied' | 'unchanged' | 'conflict' | 'partial';
	planId: string;
	writes?: number;
	target?: string;
	conflicts?: PlanEntry[];
	completed?: string[];
	inFlight?: string;
	/** Completion scan mismatches; user bytes are never rolled back. */
	changed?: { path: string; expected: string | null; actual: string | null }[];
	error?: string;
	recovery?: string;
	recoveredManifest?: boolean;
}
export interface VerifyResult {
	status: 'verified' | 'changed';
	version: string;
	source: string;
	results: { path: string; expected: string; actual: string | null }[];
}
