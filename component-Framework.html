按钮与多文件 ​

点击按钮增加计数。语言切换后，从对应 SFC 重新运行示例；状态会重置。

按钮与多文件示例

展开源码
vue
<template>
	<div class="framework-button-example">
		<CirclePlus aria-hidden="true" width="20" height="20" />
		<AsamButton @click="count++">增加计数</AsamButton>
		<Status :count="count" />
	</div>
</template>

<script lang="ts" setup>
import { ref } from 'vue';
import { AsamButton } from '@asam-ui/ui';
import Status from './Status.vue';
import { CirclePlus } from '@asam-ui/icons-vue';

defineOptions({ name: 'FrameworkButtonDemo' });

const count = ref(0);
</script>

<style scoped>
.framework-button-example {
	display: flex;
	flex-wrap: wrap;
	align-items: center;
	gap: var(--asam-spacing-md, 16px);
}
</style>
vue
<template>
	<output aria-live="polite">{{ formatCount(count) }}</output>
</template>

<script lang="ts" setup>

defineOptions({ name: 'FrameworkCountStatus' });

defineProps<{ count: number }>();
</script>
