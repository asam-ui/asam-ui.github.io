使用约定 ​

v-model 使用 Date；range 约束实际支持的显示范围，controllerType/formatter 设置控制和格式。date-cell 插槽接收 data，header 插槽及实例 selectDate 等能力以公开契约为准。日历不是字符串输入控件，不自动查询日程。

固定 2.14.7 完整用法 ​

下面将官方完整用途以 ASAM 公开组件和图标入口重写，TS/JS 均可直接运行。DatePicker 示例未指定 valueFormat 时显式设为空字符串，保留官方 Date 模型；ASAM 默认仍为日期文本。

import dayjs from 'dayjs';

interface LocaleSelection {
	locale: string;
	order: number;
}
interface LocaleSession {
	baseline: string;
	lastWritten: string;
	sequence: number;
	selections: Map<symbol, LocaleSelection>;
}
declare global {
	// Application example state is shared by TS/JS copies, keyed by the public Day.js object.
	var asamCalendarAdvancedExampleLocalesV1: WeakMap<typeof dayjs, LocaleSession> | undefined;
}

/** Own only the example's global date-locale changes; no native instance or private API. */
export function createCalendarDateLocaleSession() {
	const sessions = (globalThis.asamCalendarAdvancedExampleLocalesV1 ??= new WeakMap());
	let session = sessions.get(dayjs);
	if (!session) {
		const baseline = dayjs.locale();
		session = { baseline, lastWritten: baseline, sequence: 0, selections: new Map() };
		sessions.set(dayjs, session);
	}
	const state = session;
	const token = Symbol('calendar-example');
	let released = false;
	const observeExternalLocale = () => {
		const current = dayjs.locale();
		if (current !== state.lastWritten) state.baseline = current;
	};
	const write = (locale: string) => {
		state.lastWritten = dayjs.locale(locale);
	};
	return {
		select(locale: 'zh-cn' | 'en') {
			if (released) return;
			observeExternalLocale();
			state.selections.set(token, { locale, order: ++state.sequence });
			write(locale);
		},
		release() {
			if (released) return;
			released = true;
			observeExternalLocale();
			state.selections.delete(token);
			const remaining = [...state.selections.values()].sort((a, b) => b.order - a.order)[0];
			write(remaining?.locale ?? state.baseline);
			if (!state.selections.size) sessions.delete(dayjs);
		}
	};
}
