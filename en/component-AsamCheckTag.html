Semantic colors ​

Example: semantic-colors

Show source
Example source ZIPComplete project ZIP
vue
<template>
	<div class="colors">
		<AsamCheckTag
			v-for="entry in types"
			:key="entry.value"
			:type="entry.value"
			:checked="selected.includes(entry.value)"
			@update:checked="toggle(entry.value, $event)"
		>
			{{ entry.label }}（{{ entry.value }}）
		</AsamCheckTag>
	</div>
</template>

<script lang="ts" setup>
import { ref } from 'vue';
import { AsamCheckTag } from '@asam-ui/ui';
import type { AsamCheckTagProps } from '@asam-ui/ui';

const types = [
	{ value: 'primary', label: '主要' },
	{ value: 'success', label: '成功' },
	{ value: 'warning', label: '警告' },
	{ value: 'danger', label: '危险' },
	{ value: 'info', label: '信息' }
] satisfies { value: NonNullable<AsamCheckTagProps['type']>; label: string }[];
const selected = ref(types.map(({ value }) => value));
function toggle(type: (typeof types)[number]['value'], checked: boolean) {
	selected.value = checked
		? [...selected.value, type]
		: selected.value.filter((value) => value !== type);
}
</script>

Usage guidelines ​

Use checked or v-model:checked to control selection. The change event reports the new state. Set disabled to prevent selection and type to set the semantic color. Put the label in the default slot. The parent manages selection state and any resulting filtering.
