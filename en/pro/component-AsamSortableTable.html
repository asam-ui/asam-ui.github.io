Sortable editable table ​

Example data, control text, messages and source comments may be in Chinese. Viewer controls and API headings use English. Untranslated API descriptions are marked.

AsamSortableTable lets users reorder rows and columns independently and can also support editing cells.

When to use it ​

Row and column reordering are available by default. Enable editable to add cell editing. Use verifyCell to check an unknown write outcome. The public verifyCellResult method reports the verification result. blocked means verification did not start, and cancelled means it was invalidated.

Reorder rows ​

Drag the row handle or use the keyboard to change the row order.

vue
<template>
	<div class="pro-example-stack">
		<p>{{ instructions[mode] }}</p>
		<AsamSortableTable
			v-model="rows"
			v-model:column-order="columnOrder"
			row-key="id"
			:columns="columns"
			:row-sortable="mode === 'rows'"
			:column-sortable="mode === 'columns'"
			:editable="false"
			v-model:border="border"
			@header-dragend="showWidthChange"
		/>
		<p v-if="mode === 'width'" role="status">{{ widthChange }}</p>
		<p v-else>{{ orderSummary }}</p>
	</div>
</template>

<script setup lang="ts">
import { computed, shallowRef } from 'vue';
import { AsamSortableTable } from '@asam-ui/pro/components';
import type { AsamEditableColumn } from '@asam-ui/pro/components';

const props = defineProps<{ mode: 'rows' | 'columns' | 'width' }>();
type Row = { id: number; name: string; owner: string; status: string };
const instructions = {
	rows: '聚焦行首手柄后，可按 Alt + ↑ 或 Alt + ↓ 调整位置。',
	columns: '聚焦列标题后，可按 Alt + ← 或 Alt + → 调整位置。',
	width: '将鼠标移到可调整列的表头右边界，出现双向箭头后拖动。'
};
const rows = shallowRef<Row[]>([
	{ id: 1, name: '供应链协同平台', owner: '陈晨', status: '进行中' },
	{ id: 2, name: '园区设备巡检', owner: '林悦', status: '待启动' },
	{ id: 3, name: '研发知识库升级', owner: '张远', status: '已完成' },
	{ id: 4, name: '客户服务工作台', owner: '陈晨', status: '进行中' }
]);
const columnOrder = shallowRef(['id', 'name', 'owner', 'status']);
const columnDefinitions: AsamEditableColumn<Row>[] = [
	{ key: 'id', label: '项目编号', width: 92, fixed: 'left', draggable: false },
	{ key: 'name', label: '项目名称', minWidth: 220 },
	{ key: 'owner', label: '负责人', minWidth: 140 },
	{ key: 'status', label: '状态', minWidth: 140 }
];
const columns = computed(() =>
	columnDefinitions.map((column) => ({
		...column,
		resizable: props.mode === 'width' && column.key !== 'id'
	}))
);
const columnLabels = new Map<string, string>(
	columnDefinitions.map(({ key, label }) => [key, label])
);
const orderSummary = computed(() =>
	props.mode === 'rows'
		? '当前项目顺序：' + rows.value.map((row) => row.id).join(' → ')
		: '当前列顺序：' + columnOrder.value.map((key) => columnLabels.get(key)).join(' → ')
);
const border = shallowRef(props.mode === 'width');
const widthChange = shallowRef('');
function showWidthChange(newWidth: number, oldWidth: number) {
	widthChange.value = `列宽已从 ${Math.round(oldWidth)} px 调整到 ${Math.round(newWidth)} px。`;
}
</script>
