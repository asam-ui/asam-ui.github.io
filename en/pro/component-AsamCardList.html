Card list ​

Example data, control text, messages and source comments may be in Chinese. Viewer controls and API headings use English. Untranslated API descriptions are marked.

AsamCardList presents list results as cards. Use a grid or masonry layout and select records within the current page when the workflow requires it.

Render card content ​

Map each record to a title, cover, and summary. The operation slot can show a last-updated date and a link to details at the bottom of a card. Keep record identity stable when items are sorted or moved.

Use the Card list page template to try search, selection, editing, and details together. The card list component can also be embedded in an existing list workflow.

API and running examples ​

See the generated CardList API and demonstrations for layout, content mapping, and operation slots.

	AsamDrawer,
	AsamInfiniteScroll as vAsamInfiniteScroll,

} from '@asam-ui/ui';

import { MoreFilled } from '@asam-ui/icons-vue';

	AsamDisplayCard,

	createAsamNumberField,
	useAsamListController
} from '@asam-ui/pro/components';

	AsamFieldDefinition,
	AsamFieldModel,
	AsamListController
} from '@asam-ui/pro/components';
import { computed, onBeforeUnmount, onMounted, shallowRef, watch } from 'vue';

defineProps<{ layout: 'grid' | 'masonry' }>();

type Row = { id: number; name: string; status: string; description: string; updatedAt: string };

const cardTypes = {
	projects: '项目列表',
	all: '全部卡片类型',
	status: '状态卡片',
	task: '任务卡片',
	statistic: '指标卡片'
};
const presets = {
	small: 'small',
	default: 'default',
	big: 'big',
	large: 'large',
	custom: '自定义'
};
const states = { ready: '正常', loading: '加载中', empty: '无结果', error: '读取失败' };
const defaultGeometry = {
	minWidth: 240,
	maxWidth: 360,
	minHeight: 0,
	padding: 16,
	titleFontSize: 16,
	bodyFontSize: 13,
	contentGap: 10
};
const defaultQuery = {
	cardType: 'projects',
	columnMode: 'auto',
	preset: 'default',
	gap: 16,
	state: 'ready',
	limitWidth: 'auto',
	...defaultGeometry
} as const;
const draft = shallowRef<AsamFieldModel>({ ...defaultQuery });
const cardType = shallowRef<keyof typeof cardTypes>(defaultQuery.cardType);
const columnMode = shallowRef<'auto' | 2 | 3 | 4>(defaultQuery.columnMode);
const cardSize = shallowRef<AsamCardSize>(defaultQuery.preset);
const state = shallowRef<keyof typeof states>(defaultQuery.state);
const gap = shallowRef<number>(defaultQuery.gap);
const previewWidth = shallowRef<number>();
const previewHost = shallowRef<HTMLElement>();
const availableWidth = shallowRef(0);
let dragOrigin: { pointerId: number; x: number; width: number } | undefined;
let hostObserver: ResizeObserver | undefined;
const detailLabel = (name: string) => ({ 'aria-label': '查看详情：' + name });

function choiceField(
	key: string,
	label: string,
	options: readonly { value: string | number; label: string }[]
): AsamFieldDefinition {
	return {
		key,
		label,
		kind: 'single',
		source: defineAsamOptions().static({ options }),
		modes: { query: { componentProps: { clearable: false } } }
	};
}
function labeledChoices(labels: Readonly<Record<string, string>>) {
	return Object.entries(labels).map(([value, label]) => ({ value, label }));
}
function numberField(key: string, label: string, min: number, max: number) {
	return createAsamNumberField({
		key,
		label,
		min,
		max,
		modes: { query: { componentProps: { min, max } } }
	});
}
function numericValue(value: unknown, fallback: number) {
	return typeof value === 'number' ? value : fallback;
}
function isChoice<T extends string>(value: unknown, labels: Record<T, string>): value is T {
	return typeof value === 'string' && Object.hasOwn(labels, value);
}

const baseFields = [
	choiceField('cardType', '卡片类型', labeledChoices(cardTypes)),
	choiceField('columnMode', '列数', [
		{ value: 'auto', label: '自动' },
		{ value: 2, label: '两列' },
		{ value: 3, label: '三列' },
		{ value: 4, label: '四列' }
	]),
	choiceField('preset', '卡片尺寸', labeledChoices(presets)),
	numberField('gap', '间距', 0, 48)
];
const stateField = choiceField('state', '内容状态', labeledChoices(states));
const widthFields = [
	numberField('minWidth', '最小卡宽', 120, 1200),
	choiceField('limitWidth', '限制最大卡宽', [
		{ value: 'auto', label: '不限' },
		{ value: 'limited', label: '限制' }
	])
];
const geometryFields = [
	numberField('minHeight', '最小高度', 0, 900),
	numberField('padding', '内边距', 0, 60),
	numberField('titleFontSize', '标题字号', 12, 30),
	numberField('bodyFontSize', '正文字号', 12, 24),
	numberField('contentGap', '内容间距', 0, 40)
];
const projectQuery = computed(() => draft.value.cardType === 'projects');
const customSize = computed(() => draft.value.preset === 'custom');
const widthLimit = computed(() => draft.value.limitWidth === 'limited');
const minimumWidth = computed(() =>
	Math.min(numericValue(draft.value.minWidth, defaultGeometry.minWidth), 1200)
);
const fields = computed<readonly AsamFieldDefinition[]>(() => {
	const visibleFields = [...baseFields];
	if (projectQuery.value) visibleFields.push(stateField);
	if (customSize.value) {
		visibleFields.push(...widthFields);
		if (widthLimit.value)
			visibleFields.push(numberField('maxWidth', '最大卡宽', minimumWidth.value, 1600));
		visibleFields.push(...geometryFields);
	}
	return visibleFields;
});

onMounted(() => {
	const host = previewHost.value;
	if (!host) return;
	const measure = () => (availableWidth.value = host.clientWidth);
	measure();
	if (typeof ResizeObserver !== 'undefined') {
		hostObserver = new ResizeObserver(measure);
		hostObserver.observe(host);
	}
});

function setPreviewWidth(width: number) {
	const available = availableWidth.value;
	if (!available) return;
	const minimum = Math.min(240, available);
	previewWidth.value = width >= available ? undefined : Math.max(minimum, width);
}
function currentPreviewWidth() {
	return Math.min(previewWidth.value ?? availableWidth.value, availableWidth.value);
}
function beginResize(event: PointerEvent) {
	if (event.button !== 0 || dragOrigin || !(event.currentTarget instanceof HTMLElement)) return;
	dragOrigin = { pointerId: event.pointerId, x: event.clientX, width: currentPreviewWidth() };
	event.currentTarget.setPointerCapture(event.pointerId);
}
function resizePreview(event: PointerEvent) {
	if (dragOrigin?.pointerId !== event.pointerId) return;
	setPreviewWidth(dragOrigin.width + event.clientX - dragOrigin.x);
}
function nudgePreview(delta: number) {
	setPreviewWidth(currentPreviewWidth() + delta);
}
const active = shallowRef<Row>();
const readMode = shallowRef<'first' | 'append'>('first');
const coverThrough = shallowRef(0);
let coverTimer: ReturnType<typeof setTimeout> | undefined;
let failNextRead = false;
const createRow = (index: number): Row => ({
	id: index + 1,
	name: '项目 ' + (index + 1),
	updatedAt: '2026-11-' + String((index % 11) + 1).padStart(2, '0'),
	status: ['进行中', '已完成', '待确认'][index % 3]!,
	description: [
		'项目范围已确认。',
		'正在整理阶段成果，涉及范围核对、进度跟踪和后续验收。',
		'等待下一轮评审。团队已整理需求、交互、实现及用户复测记录，确认后安排后续交付。'
	][index % 3]!
});
const records = Array.from({ length: 37 }, (_, index) => createRow(index));
function waitForResponse(delay: number, signal: AbortSignal): Promise<void> {
	return new Promise((resolve, reject) => {
		signal.throwIfAborted();
		const cancel = () => {
			clearTimeout(timer);
			reject(signal.reason);
		};
		const timer = setTimeout(() => {
			signal.removeEventListener('abort', cancel);
			resolve();
		}, delay);
		signal.addEventListener('abort', cancel, { once: true });
	});
}

const runtime = createAsamDataRuntime();
const controller: AsamListController<Row> = useAsamListController<Row>({
	runtime,
	rowKey: 'id',
	fields: [],
	initial: {},
	api: async ({ pageSize }, { signal }) => {
		const current = state.value;
		const append = readMode.value === 'append';
		const previous = append ? controller.page.data.value.list : [];
		const offset = previous.length;
		await waitForResponse(append ? 550 : current === 'loading' ? 1000 : 220, signal);
		signal.throwIfAborted();
		if (failNextRead) {
			failNextRead = false;
			throw new Error('示例读取失败');
		}
		const rows = current === 'empty' ? [] : records;
		return {
			list: [...previous, ...rows.slice(offset, offset + pageSize)],
			total: rows.length
		};
	}
});

const page = controller.page;
const loadedCount = computed(() => page.data.value.list.length);
const hasMore = computed(() => loadedCount.value < page.data.value.total);
const scrollDisabled = computed(
	() => cardType.value !== 'projects' || !controller.resultReady.value || !hasMore.value
);
const appendFailed = computed(() => readMode.value === 'append' && !!page.error.value);
const scrollStatus = computed(() => {
	if (readMode.value === 'append' && page.loading.value) return '正在加载更多项目…';
	if (!controller.resultReady.value || !loadedCount.value) return '';
	return hasMore.value ? '继续滚动加载更多项目' : '已加载全部项目';
});

function loadMore() {
	if (scrollDisabled.value) return;
	readMode.value = 'append';
	// 重读当前结果快照，保留已选行，追加来源的下一批记录。
	void page.reload({ force: true });
}

function retry() {
	if (page.loading.value || !page.error.value) return;
	void page.reload({ force: true });
}

watch(page.data, ({ list }) => {
	clearTimeout(coverTimer);
	const lastId = list.at(-1)?.id ?? 0;
	if (lastId > coverThrough.value) {
		coverTimer = setTimeout(() => (coverThrough.value = lastId), 650);
	}
});

function applyQuery(query: AsamFieldModel) {
	const previousType = cardType.value;
	cardType.value = isChoice(query.cardType, cardTypes) ? query.cardType : defaultQuery.cardType;
	const columns = query.columnMode;
	columnMode.value = columns === 2 || columns === 3 || columns === 4 ? columns : 'auto';
	gap.value = numericValue(query.gap, defaultQuery.gap);

	const preset = isChoice(query.preset, presets) ? query.preset : defaultQuery.preset;
	if (preset === 'custom') {
		const minWidth = numericValue(query.minWidth, defaultGeometry.minWidth);
		cardSize.value = {
			minWidth,
			maxWidth:
				query.limitWidth === 'limited'
					? Math.max(minWidth, numericValue(query.maxWidth, defaultGeometry.maxWidth))
					: undefined,
			minHeight: numericValue(query.minHeight, defaultGeometry.minHeight),
			padding: numericValue(query.padding, defaultGeometry.padding),
			titleFontSize: numericValue(query.titleFontSize, defaultGeometry.titleFontSize),
			bodyFontSize: numericValue(query.bodyFontSize, defaultGeometry.bodyFontSize),
			contentGap: numericValue(query.contentGap, defaultGeometry.contentGap)
		};
	} else {
		cardSize.value = preset;
	}

	if (cardType.value !== 'projects') {
		if (previousType === 'projects') {
			page.cancel();
			clearTimeout(coverTimer);
			active.value = undefined;
		}
		return;
	}
	const nextState = isChoice(query.state, states) ? query.state : defaultQuery.state;
	if (
		previousType === 'projects' &&
		state.value === nextState &&
		nextState !== 'error' &&
		nextState !== 'loading'
	)
		return;
	page.cancel();
	clearTimeout(coverTimer);
	coverThrough.value = 0;
	active.value = undefined;
	readMode.value = 'first';
	state.value = nextState;
	failNextRead = nextState === 'error';
	void controller.submit();
}
const media =
	'data:image/svg+xml,' +
	encodeURIComponent(
		'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 400 160"><rect width="400" height="160" fill="#e4f0ff"/><path d="M0 152l108-104 72 66 62-46 158 92H0Z" fill="#73a9ec"/><circle cx="290" cy="55" r="22" fill="#ffd089"/></svg>'
	);
const content = (row: Row) => ({
	title: row.name,
	description: row.description,
	cover:
		row.id % 3 === 1 || (row.id <= coverThrough.value && row.id % 3 === 2)
			? { src: media, alt: row.name + '封面' }
			: undefined
});
const formatDate = new Intl.DateTimeFormat('zh-CN', {
	month: 'long',
	day: 'numeric',
	timeZone: 'UTC'
});
const updatedLabel = (value: string) => formatDate.format(new Date(value)) + '更新';
onBeforeUnmount(() => {
	hostObserver?.disconnect();
	clearTimeout(coverTimer);
	runtime.invalidate();
});
function close(open: boolean) {
	if (!open) active.value = undefined;
}
</script>
