Editable table ​

Example data, control text, messages and source comments may be in Chinese. Viewer controls and API headings use English. Untranslated API descriptions are marked.

AsamEditableTable supports editing fields in table cells and synchronizes the current page of records through v-model.

When to use it ​

Configure a column with field or use the shared fields configuration. Provide persistCell to save a cell asynchronously. If the result is unknown, use verifyCell to check whether the original write was applied. Verification checks the existing operation without issuing another save. If the check confirms that the write was never applied, the host decides whether to restore the draft.

Four editing modes ​

The examples use AsamTableEditingHost for single-row, multiple-row, instant, and cell editing. The host shares the list, field, and write contracts across the modes. Cancel and leave are explicit actions, and unknown write outcomes require verification. The API reference includes the host API.

Edit one row ​

Start editing with the row's Edit action. Change its fields, then save or cancel.

	AsamTableEditingPresentation

defineProps<{ mode: AsamTableEditingPresentation }>();

interface Row {
	id: number;
	name: string;
	status: string;
	budget: number;
}

const statuses = ['待启动', '进行中', '已完成'];
const records: Row[] = [
	{ id: 1, name: '供应链协同平台', status: '进行中', budget: 48000 },
	{ id: 2, name: '园区设备巡检', status: '待启动', budget: 55500 },
	{ id: 3, name: '研发知识库升级', status: '进行中', budget: 63000 },
	{ id: 4, name: '客户服务工作台', status: '已完成', budget: 70500 }
];
const fields: AsamFieldDefinition[] = [
	{
		key: 'name',
		label: '项目名称',
		kind: 'text',
		modes: {
			edit: { rules: [{ required: true, whitespace: true, message: '请输入项目名称' }] }
		}
	},
	{
		key: 'status',
		label: '状态',
		kind: 'single',
		source: defineAsamOptions().static({
			options: statuses.map((status) => ({ value: status, label: status }))
		})
	},
	createAsamNumberField({ key: 'budget', label: '预算（元）', min: 0, required: true })
];
const columns: AsamEditableColumn<Row>[] = [
	{ key: 'id', label: '编号', width: 112, draggable: false },
	{ key: 'name', label: '项目名称', minWidth: 220 },
	{ key: 'status', label: '状态', minWidth: 120 },
	{ key: 'budget', label: '预算（元）', minWidth: 145 }
];

const runtime = createAsamDataRuntime();
const list = useAsamListController<Row>({
	runtime,
	rowKey: 'id',
	fields: [],
	initial: {},
	api: async () => ({ list: records.map((row) => ({ ...row })), total: records.length })
});

function confirmDiscard() {
	return AsamMessageBox.confirm('放弃尚未保存的内容？', '取消编辑', {
		confirmButtonText: '放弃修改',
		cancelButtonText: '继续编辑'
	}).then(
		() => true,
		() => false
	);
}

const editing = useAsamTableEditing<Row>({
	list,
	runtime,
	coordinator: createAsamWriteCoordinator(),
	fields,
	equals: { budget: Object.is },
	confirmDiscard,
	save: async (payload, operation) => {
		const row = records.find((item) => item.id === operation.target.id);
		if (!row) return { status: 'failure', message: '记录已不存在' };
		Object.assign(row, payload);
		return { status: 'success' };
	},
	saveOrder: async () => ({ status: 'failure', message: '本例只保存字段变更' })
});

async function persistCell(change: AsamCellChange<Row>) {
	if (!change.isCurrent()) return { status: 'failure' as const, message: '编辑已失效' };
	const row = records.find((item) => item.id === change.rowKey);
	if (!row) return { status: 'failure' as const, message: '记录已不存在' };
	Object.assign(row, { [change.columnKey]: change.value });
	return { status: 'success' as const };
}

onBeforeUnmount(() => {
	editing.dispose();
	runtime.invalidate();
});
</script>
