Usage guidelines ​

Pass a string or an array of strings to content, or use image for a graphic. Configure the layout with font, rotate, width, height, gap, offset, and zIndex. Put the covered content in the default slot and give the container measurable dimensions. The example sets the watermark text to ASAM. Access control remains the responsibility of the host application.

<script lang="ts" setup>
defineOptions({ name: 'AsamWatermarkCapCustomDemo' });
import {
	AsamImage,
	AsamButton,
	AsamWatermark,
	AsamBaseForm,
	AsamFormItem,
	AsamInput,
	AsamColorPicker,
	AsamSlider,
	AsamSpace,
	AsamInputNumber
} from '@asam-ui/ui';

import { computed, reactive, watch } from 'vue';
import { ref } from 'vue';
const isDark = ref(false);

const gap = reactive<[number, number]>([100, 100]);
const offset = reactive<[number | undefined, number | undefined]>([undefined, undefined]);

const config = reactive({
	content: 'ASAM UI',
	font: {
		fontSize: 16,
		color: 'rgba(0, 0, 0, 0.15)'
	},
	zIndex: -1,
	rotate: -22,
	gap
});

const watermarkOffset = computed<[number, number] | undefined>(() => {
	const [left, top] = offset;
	return left === undefined && top === undefined
		? undefined
		: [left ?? config.gap[0] / 2, top ?? config.gap[1] / 2];
});

watch(
	isDark,
	(value) => {
		config.font.color = value ? 'rgba(255, 255, 255, .15)' : 'rgba(0, 0, 0, .15)';
	},
	{ immediate: true }
);
</script>
