Data form ​

Example data, control text, messages and source comments may be in Chinese. Viewer controls and API headings use English. Untranslated API descriptions are marked.

Use AsamForm to describe fields, dependencies and data sources in a typed schema. For fields specified directly in the template, use Base form.

Schema and custom controls ​

createAsamSchema checks each field name, its model type and its control. Built-in controls are already registered. Use defineAsamControl to add a new custom: control to registry.

The registered date picker, time picker and time select controls use a string | null field contract. Choose the field representation before building the schema. Try the scalar date and time fields.

Dependencies and visibility ​

Declare prerequisite fields with dependsOn. Use resolveDataSource when the source must be derived from the read-only model. Configure visibility and validation according to the field’s role.

Field slots and the form instance ​

A field-{name} slot receives field, model, value, update and updateValue. Call the supplied update function to write the field back to the form.

Layout defaults ​

The defaults include inline layout, automatic label width, visible labels, and Search and Reset actions. Override these settings when the form needs a different layout or actions.

Base form behavior ​

See Base form for compatible form behavior, field validation and size inheritance. Data form owns its model update, schema and data-source rules.

import type { AsamFormExpose, AsamValue, AsamOptionsSource } from '@asam-ui/ui';
interface Model {
	region: string;
	note: string;
	member: AsamValue | null;
}
interface Member {
	label: string;
	value: string;
}
const model = ref<Model>({ region: 'east', note: '', member: null });
const submitted = ref<Model | null>(null);
const form = ref<AsamFormExpose<Model>>();
const retainDependency = ref(false);
const retainHidden = ref(false);
const hideNote = ref(false);
const disableNote = ref(false);
const booleanHidden = ref(false);
const booleanDisabled = ref(true);
const fixedSourceMode = ref(false);
const defaultMemo = ref(false);
const extraRule = ref(false);
const alternateMemberPresentation = ref(false);
const memberContext = ref({
	present: false,
	propMatches: false,
	listContainsField: false,
	count: 0
});
const requiredMember = ref(false);
const extraMemberRule = ref(false);
const hideMember = ref(false);
const disableMember = ref(false);
const retainMemberHide = ref(true);
const defaultMember = ref(false);
const booleanMemberHidden = ref(false);
const booleanMemberDisabled = ref(true);
const validation = ref('未执行');
const validateCount = ref(0);
const resetCount = ref(0);
const fields = ref<{ count: number; hasRegion: boolean }>({ count: 0, hasRegion: false });
const sources = new Map<string, AsamOptionsSource<string, Member, 'api'>>();
function sourceFor(region: string) {
	if (!region) return undefined;
	return ensureSource(region);
}
function ensureSource(region: string) {
	let source = sources.get(region);
	if (!source) {
		source = defineAsamOptions<string, Member>().api({
			api: async (_params: undefined, context) => {
				await new Promise<void>((done) => setTimeout(done, 100));
				context.signal.throwIfAborted();
				return [{ label: region + ' Alpha', value: region + '-alpha' }];
			},
			params: undefined,
			transform: (rows) => rows,
			fieldNames: { label: 'label', value: 'value' }
		});
		sources.set(region, source);
	}
	return source;
}
const builder = createAsamSchema<Model>()(asamFormControls);
const schema = computed(() => {

		calls = ref(0),
		refreshes = ref(0);
	const api = async () => {
		calls.value++;
		if (failing.value) throw Error('演示请求暂不可用');
		return [
			{ id: 1, value: 'ready', label: '准备' },
			{ id: 2, value: 'running', label: '进行' }
		] satisfies readonly Item[];
	};

		dict: {
			get: async (
				_type: string,
				context: Parameters<NonNullable<AsamDataProvider['dict']>['get']>[1]
			) => {
				context.signal.throwIfAborted();
				return api();
			},
			refresh: async (context) => {
				context.signal.throwIfAborted();
				refreshes.value++;
			}
		}
	};
	return { failing, calls, refreshes, api, provider };
}

	};

		dict: {
			get: async (_type, context) => {
				context.signal.throwIfAborted();
				return api();
			},
			refresh: async (context) => {
				context.signal.throwIfAborted();

			}
		}
	};
