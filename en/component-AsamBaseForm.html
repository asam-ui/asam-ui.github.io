Base form ​

Example data, control text, messages and source comments may be in Chinese. Viewer controls and API headings use English. Untranslated API descriptions are marked.

When to use it ​

Use AsamBaseForm when your template declares the fields. Pass a reactive model managed by the caller, and bind each Form item to its field using prop. Use Data form for a schema that generates fields.

Validation, native submit and reset ​

	<div class="example">
		<AsamBaseForm

			v-bind="{ ...{ 'label-position': 'top' }, ...{ model: model, rules: rules } }"
			@submit.prevent="submit"

</template>

			:show-message="showMessage"
			status-icon
			scroll-to-error
			:scroll-into-view-options="{ behavior: 'smooth', block: 'center' }"
			@validate="(prop, valid, message) => events.push(`${prop}:${valid}:${message}`)"
			@submit.prevent="validate"
		>
			<AsamFormSection title="基础资料" description="异步规则与布局组合由使用方定义">
				<AsamFormGrid :columns="2" :tablet-columns="2" :mobile-columns="1">
					<AsamFormItem label="账号" prop="account" help="输入 taken 将触发异步占用检查">
						<AsamInput v-model="model.account" aria-label="账号" />
					</AsamFormItem>
					<AsamFormItem label="确认账号" prop="confirm" label-position="top">
						<AsamInput v-model="model.confirm" aria-label="确认账号" />
					</AsamFormItem>
				</AsamFormGrid>
			</AsamFormSection>
			<AsamFormActions
				:saving="saving"
				submit-native-type="submit"
				@cancel="form?.resetFields()"
			>
				<template #leading><span>原生 submit 只在表单处理一次</span></template>
			</AsamFormActions>
		</AsamBaseForm>
		<div class="controls">
			<AsamButton type="default" @click="setInitial">设置新的初始值</AsamButton>
			<AsamButton type="default" @click="form?.resetFields()">恢复初始值</AsamButton>
			<AsamButton type="default" @click="form?.clearValidate()">清除全部提示</AsamButton>
			<AsamButton type="primary" @click="inspect">查看注册字段</AsamButton>
		</div>
		<output>{{ result }}</output>
		<pre>{{ JSON.stringify(model) }}</pre>

			</AsamCollapseItem>
		</AsamCollapse>
	</div>

<script lang="ts" setup>
defineOptions({ name: 'AsamBaseFormAdvancedExample' });
import { reactive, ref } from 'vue';
import {
	AsamCollapse,
	AsamCollapseItem,
	AsamSelect,
	AsamOption,
	AsamCheckboxItem,
	AsamBaseForm,
	AsamFormItem,
	AsamInput,
	AsamFormActions,
	AsamFormSection,
	AsamFormGrid,
	AsamButton
} from '@asam-ui/ui';
import type { AsamBaseFormExpose, AsamBaseFormProps } from '@asam-ui/ui';
const form = ref<AsamBaseFormExpose>();
const model = reactive({ account: '', confirm: '' });
const position = ref<'left' | 'right' | 'top'>('right');
const inline = ref(false);
const disabled = ref(false);
const showMessage = ref(true);
const saving = ref(false);
const result = ref('尚未校验');
const events = ref<string[]>([]);
const rules: AsamBaseFormProps['rules'] = {
	account: [
		{ required: true, message: '请输入账号', trigger: 'blur' },
		{
			async validator(_rule, value) {
				await new Promise((resolve) => setTimeout(resolve, 100));
				if (value === 'taken') throw new Error('账号已占用');
			},
			trigger: 'blur'
		}
	],
	confirm: [
		{
			validator(_rule, value, callback) {
				callback(
					value === model.account && value ? undefined : new Error('两次账号需一致')
				);
			},
			trigger: 'blur'
		}
	]
};
