Tag ​

Example data, control text, messages and source comments may be in Chinese. Viewer controls and API headings use English. Untranslated API descriptions are marked.

When to use it ​

Use AsamTag for a short status, category or removable label.

Semantic colors ​

type defaults to primary. The other types are secondary, gradient, success, warning, danger, and info. In light and dark effects, the gradient tag uses the same fill and text color as a gradient button. In the plain effect it uses the primary color for text and outline. Compare tag colors.

Appearance and interaction ​

Use effect to choose the visual treatment. Add round for a pill shape or closable for a close control. Handle the close event in the owner of the tag list to remove the corresponding item.

Editable tag lists ​

Example: cap-editable

Show source
LicenseExample source ZIPComplete project ZIP
vue
<template>
	<div class="flex gap-2">
		<AsamTag
			v-for="tag in dynamicTags"
			:key="tag.id"
			:type="tag.type"
			closable
			:disable-transitions="false"
			@close="handleClose(tag.id)"
		>
			{{ tag.label }}
		</AsamTag>
		<AsamInput
			v-if="inputVisible"
			ref="InputRef"
			:model-value="inputValue"
			class="w-20"
			size="small"
			v-bind="domEnterAttrs(handleInputConfirm)"
			@update:model-value="
				(value) => {
					if (typeof value === 'string') inputValue = value;
				}
			"
			@blur="handleInputConfirm"
		/>
		<!-- prettier-ignore -->
		<AsamButton type="primary"
			v-else
			class="button-new-tag"
			size="small"
			@click="showInput"
		>
			+ 新建标签
		</AsamButton>
	</div>
</template>

<script lang="ts" setup>
import type { HTMLAttributes } from 'vue';
defineOptions({ name: 'AsamTagCapEditableDemo' });
import { AsamTag, AsamInput, AsamButton } from '@asam-ui/ui';

import { nextTick, ref } from 'vue';

import type { AsamInputExpose, AsamTagType } from '@asam-ui/ui';

interface DemoTag {
	id: number;
	label: string;
	type: AsamTagType;
}

const inputValue = ref('');
const dynamicTags = ref<DemoTag[]>([
	{ id: 1, label: '标签一', type: 'primary' },
	{ id: 2, label: '次要', type: 'secondary' },
	{ id: 3, label: '渐变', type: 'gradient' }
]);
let nextTagId = 4;
const inputVisible = ref(false);
const InputRef = ref<AsamInputExpose>();

const handleClose = (id: number) => {
	dynamicTags.value = dynamicTags.value.filter((tag) => tag.id !== id);
};

const showInput = () => {
	inputVisible.value = true;
	nextTick(() => {
		InputRef.value!.input!.focus();
	});
};

const handleInputConfirm = () => {
	if (inputValue.value) {
		dynamicTags.value.push({ id: nextTagId++, label: inputValue.value, type: 'primary' });
	}
	inputVisible.value = false;
	inputValue.value = '';
};

const domEnterAttrs = (handler: () => void): Pick<HTMLAttributes, 'class' | 'onKeyup'> => ({
	class: 'ui032-enter-input',
	onKeyup: (event) => {
		if (event.key === 'Enter') handler();
	}
});
</script>
