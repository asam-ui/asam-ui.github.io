		<AsamTable

		>
			<!-- @vue-generic {Row} -->

				<template #expand="{ expanded, expandable }">
					<span>{{ expandable ? (expanded ? '收起' : '展开') : '不可展开' }}</span>
				</template>
				<template #default="{ row }">{{ row.name }} 的完整详情</template>
			</AsamTableColumn>
			<!-- @vue-generic {Row} -->
			<AsamTableColumn prop="name" label="名称" sortable>
				<template #header="{ column, $index }">{{ $index }} · {{ column.label }}</template>
				<template #default="{ row, $index }">{{ $index + 1 }} · {{ row.name }}</template>
			</AsamTableColumn>
			<!-- @vue-generic {Row} -->
			<AsamTableColumn
				prop="group"
				label="分类"
				column-key="group"
				:filters="[
					{ text: '甲', value: 'a' },
					{ text: '乙', value: 'b' }
				]"
				:filtered-value="filtered"
				:filter-method="(value, row) => row.group === value"
			>
				<template #filter-icon="{ filterOpened }">
					<span>{{ filterOpened ? '收起筛选' : '打开筛选' }}</span>
				</template>

		</AsamTable>

			"

			清空选择
		</AsamButton>

	</div>
</template>

用法与契约补充说明 ​

契约与组合 ​

列在 AsamTable 内注册，排序、过滤、选择与树状态由父表格管理。prop / property、label、width / minWidth、fixed、sortable / sortMethod / sortBy、formatter、filters / filterMethod、filteredValue、showOverflowTooltip / tooltipFormatter 和 index 使用本入口的公开契约。列不独立接收四档 size，密度来自表格。

契约与组合 ​

公开 Props、Slots 与实例类型接受 Row 泛型。default 仅提供 row / column / $index；header 提供 column / $index；expand 图标插槽仅提供 expanded / expandable；filter-icon 仅提供 filterOpened。详情正文放在 type="expand" 列的 default，图标插槽与详情正文拥有不同载荷。原生列不声明独立命令式 expose 方法；注册与卸载由父表格完成。
